<script>
(function(){
  // Theme presets
  const THEMES = {
    pomodoro:{ bg:'#b64f5b', message:'Time to focus!', minutes:25 },
    short:{ bg:'#84666b', message:'Time for a break!', minutes:5 },
    long:{ bg:'#3e8889', message:'Time for a long break!', minutes:15 }
  };

  const tabs = document.querySelectorAll('.tab');
  const timeEl = document.getElementById('time');
  const startPauseBtn = document.getElementById('startPause');
  const skipBtn = document.getElementById('skipBtn');
  const subtitleEl = document.getElementById('subtitle');
  const tasksList = document.getElementById('tasksList');
  const addTaskBtn = document.getElementById('addTaskBtn');
  const taskForm = document.getElementById('taskForm');
  const taskInput = document.getElementById('taskInput');
  const estNumEl = document.getElementById('estNum');
  const upBtn = document.getElementById('upBtn');
  const downBtn = document.getElementById('downBtn');
  const cancelForm = document.getElementById('cancelForm');
  const saveTask = document.getElementById('saveTask');
  const summaryBar = document.getElementById('summaryBar');
  const pomosCount = document.getElementById('pomosCount');
  const finishAt = document.getElementById('finishAt');
  const finishHours = document.getElementById('finishHours');
  const finishedToast = document.getElementById('finishedToast');
  const clearTasksBtn = document.getElementById('clearTasksBtn');
  const confettiLayer = document.getElementById('confettiLayer');

  // State
  let mode = 'pomodoro';
  let timer = null;
  let remaining = THEMES[mode].minutes * 60;
  let tasks = []; // each: {id, title, est, done}
  let activeTaskId = null;

  function setTheme(themeKey){
    mode = themeKey;
    const t = THEMES[mode];
    document.documentElement.style.setProperty('--bg', t.bg);
    remaining = t.minutes * 60;
    updateTime();
    tabs.forEach(tab=>tab.classList.toggle('active', tab.dataset.mode===mode));
    subtitleEl.textContent = t.message;
    startPauseBtn.textContent = 'START';
    stopTimer();
  }

  function updateTime(){
    const m = Math.floor(remaining/60).toString().padStart(2,'0');
    const s = Math.floor(remaining%60).toString().padStart(2,'0');
    timeEl.textContent = `${m}:${s}`;
  }

  function startTimer(){
    if(timer) return;
    timer = setInterval(()=>{
      remaining--;
      if(remaining<0){
        stopTimer();
        startPauseBtn.textContent='START';
        // auto switch to short break for demo
        setTheme(mode==='pomodoro'?'short':'pomodoro');
        return;
      }
      updateTime();
    }, 1000);
  }
  function stopTimer(){
    if(timer){
      clearInterval(timer);
      timer=null;
    }
  }

  tabs.forEach(tab=>{
    tab.addEventListener('click', ()=> setTheme(tab.dataset.mode));
  });

  startPauseBtn.addEventListener('click', ()=>{
    if(timer){
      stopTimer();
      startPauseBtn.textContent='START';
    }else{
      startTimer();
      startPauseBtn.textContent='PAUSE';
    }
  });

  skipBtn.addEventListener('click', ()=>{
    // simple jump 3s for visual effect
    remaining = Math.max(0, remaining-3);
    updateTime();
  });

  // Tasks logic
  function openForm(){
    taskForm.style.display='block';
    taskInput.focus();
  }
  function closeForm(){
    taskForm.style.display='none';
    taskInput.value='';
    estNumEl.textContent='1';
  }
  addTaskBtn.addEventListener('click', openForm);
  cancelForm.addEventListener('click', closeForm);
  upBtn.addEventListener('click', ()=> estNumEl.textContent = String(Math.min(99, (+estNumEl.textContent)+1)));
  downBtn.addEventListener('click', ()=> estNumEl.textContent = String(Math.max(1, (+estNumEl.textContent)-1)));

  saveTask.addEventListener('click', ()=>{
    const title = taskInput.value.trim() || 'Untitled';
    const est = +estNumEl.textContent;
    const id = Date.now();
    tasks.unshift({id, title, est, done:0});
    activeTaskId = id;
    renderTasks();
    closeForm();
  });

  function renderTasks(){
    tasksList.innerHTML='';
    tasks.forEach((t, idx)=>{
      const row = document.createElement('div');
      row.className='task-row';
      row.innerHTML = `
        <div class="task-left">
          <span class="btn check" title="Toggle Complete">
            <img src="__PLACEHOLDER_ASSETS_BASE_DIR__/016.png" alt="check" width="18" height="18">
          </span>
          <div class="title">${escapeHtml(t.title)}</div>
        </div>
        <div class="task-right">
          <div class="count">${t.done} / ${t.est}</div>
          <div class="kebab btn">⋮</div>
        </div>
      `;
      // set active information under the timer
      if(idx===0 && activeTaskId===t.id){
        document.getElementById('currentTaskTitle').querySelector('.muted').textContent='#1';
        document.getElementById('currentTaskTitle').querySelector('#subtitle').textContent = THEMES[mode].message.replace('focus','focus!');
      }
      // toggle on check
      row.querySelector('.check').addEventListener('click', ()=>{
        t.done = (t.done < t.est) ? t.est : 0;
        renderTasks();
        checkAllDone();
      });
      // clicking the row selects it
      row.addEventListener('click', ()=>{ activeTaskId=t.id; });
      tasksList.appendChild(row);
    });
    updateSummary();
  }

  function updateSummary(){
    const total = tasks.reduce((a,t)=>a+t.est,0);
    const done = tasks.reduce((a,t)=>a+t.done,0);
    if(total===0){
      summaryBar.style.display='none';
      pomosCount.textContent='0 / 0';
      return;
    }
    summaryBar.style.display='';
    pomosCount.textContent=`${done} / ${total}`;
    // naive finish time estimate: remaining pomos * 25 minutes
    const remain = Math.max(0, total-done);
    const now = new Date();
    const minutes = remain*25;
    const finish = new Date(now.getTime()+minutes*60000);
    finishAt.textContent = finish.toTimeString().slice(0,5);
    finishHours.textContent = `(${(minutes/60).toFixed(1)}h)`;
  }

  function checkAllDone(){
    const total = tasks.reduce((a,t)=>a+t.est,0);
    const done = tasks.reduce((a,t)=>a+t.done,0);
    if(total>0 && done===total){
      finishedToast.style.display='flex';
      dropConfetti();
    }
  }

  clearTasksBtn.addEventListener('click', ()=>{
    tasks = [];
    activeTaskId=null;
    renderTasks();
    finishedToast.style.display='none';
    confettiLayer.style.display='none';
    confettiLayer.innerHTML='';
  });

  // Confetti animation
  function dropConfetti(){
    confettiLayer.style.display='block';
    confettiLayer.innerHTML='';
    const colors = ['#f94144','#f3722c','#f9c74f','#90be6d','#43aa8b','#577590','#ffd166','#e76f51','#06d6a0','#a29bfe'];
    const count = 80;
    for(let i=0;i<count;i++){
      const s = document.createElement('span');
      s.style.left = (Math.random()*100)+'%';
      s.style.top = (-Math.random()*40)+'px';
      s.style.background = colors[Math.floor(Math.random()*colors.length)];
      s.style.width = s.style.height = (6 + Math.random()*8)+'px';
      s.style.animationDuration = (900 + Math.random()*900)+'ms';
      confettiLayer.appendChild(s);
    }
    // hide later
    setTimeout(()=>{ confettiLayer.style.display='none'; confettiLayer.innerHTML=''; }, 2000);
  }

  // Utilities
  function escapeHtml(str){
    return str.replace(/[&<>"']/g, m=>({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[m]));
  }

  // Initialize
  setTheme('pomodoro');
  updateSummary();

  // Mock: open task form on load to resemble screenshots
  setTimeout(()=>openForm(), 400);
})();
</script>
</body>
