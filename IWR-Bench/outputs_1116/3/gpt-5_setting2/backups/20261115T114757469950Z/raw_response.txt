<script>
(function(){
  const timeEl = document.getElementById('time');
  const startBtn = document.getElementById('startBtn');
  const timeline = document.getElementById('timeline');
  const focusText = document.getElementById('focusText');
  const currentIndexEl = document.getElementById('currentIndex');

  const editor = document.getElementById('editor');
  const openEditor = document.getElementById('openEditor');
  const cancelEditor = document.getElementById('cancelEditor');
  const saveTask = document.getElementById('saveTask');
  const taskTitleInput = document.getElementById('taskTitle');
  const up = document.getElementById('up');
  const down = document.getElementById('down');
  const estVal = document.getElementById('estVal');
  const taskList = document.getElementById('taskList');

  const toast = document.getElementById('toast');
  const clearTasks = document.getElementById('clearTasks');

  const doneCountEl = document.getElementById('doneCount');
  const totalCountEl = document.getElementById('totalCount');
  const finishAtEl = document.getElementById('finishAt');
  const hoursEl = document.getElementById('hours');

  const confettiBox = document.getElementById('confetti');

  // State
  const DURS = { pomodoro: 25*60, short: 5*60, long: 15*60 };
  let mode = 'pomodoro';
  let left = DURS[mode];
  let timer = null;
  let running = false;

  let tasks = []; // {id, title, est, done, complete}
  let activeTaskIndex = 0;

  // Helpers
  function fmt(s){
    const m = Math.floor(s/60).toString().padStart(2,'0');
    const ss = Math.floor(s%60).toString().padStart(2,'0');
    return `${m}:${ss}`;
  }
  function setTheme(){
    document.body.classList.remove('theme-pomodoro','theme-short','theme-long');
    document.body.classList.add(mode==='pomodoro'?'theme-pomodoro':(mode==='short'?'theme-short':'theme-long'));
    focusText.textContent = mode==='pomodoro' ? 'Time to focus!' : 'Time for a break!';
  }
  function renderTime(){
    timeEl.textContent = fmt(left);
    timeline.value = Math.min(100, 100 - (left/DURS[mode])*100);
  }
  function switchMode(target){
    mode = target;
    left = DURS[mode];
    running = false;
    clearInterval(timer);
    startBtn.textContent = 'START';
    document.querySelectorAll('.mode').forEach(m => m.classList.toggle('active', m.dataset.mode===mode));
    setTheme();
    renderTime();
  }
  function tick(){
    if(left>0){ left--; renderTime(); }
    else{
      clearInterval(timer); running=false;
      startBtn.textContent = 'START';
      // If working session is over, increment pomodoro count for active task
      if(mode==='pomodoro'){
        const t = tasks[activeTaskIndex];
        if(t){
          t.done++; if(t.done>=t.est){ t.complete=true; }
          refreshTasks();
          checkAllDone();
        }
      }
      // simple confetti splash
      splashConfetti();
    }
  }
  function startPause(){
    if(running){ running=false; startBtn.textContent='START'; clearInterval(timer); }
    else{
      running=true; startBtn.textContent='PAUSE'; timer=setInterval(tick,1000);
    }
  }

  // Tasks
  function openTaskEditor(){
    editor.classList.add('active');
    taskTitleInput.value='';
    estVal.textContent='1';
    taskTitleInput.focus();
  }
  function closeTaskEditor(){ editor.classList.remove('active'); }

  up.addEventListener('click', ()=>{ estVal.textContent = Math.min(20, (+estVal.textContent)+1); });
  down.addEventListener('click', ()=>{ estVal.textContent = Math.max(1, (+estVal.textContent)-1); });

  openEditor.addEventListener('click', openTaskEditor);
  cancelEditor.addEventListener('click', closeTaskEditor);
  saveTask.addEventListener('click', ()=>{
    const title = taskTitleInput.value.trim()||'Untitled';
    const est = +estVal.textContent;
    tasks.push({id:Date.now(), title, est, done:0, complete:false});
    activeTaskIndex = tasks.length-1;
    closeTaskEditor();
    refreshTasks();
    scrollToTasks();
  });

  function taskItem(t,i){
    const div = document.createElement('div');
    div.className = 'task'+(t.complete?' done':'');
    div.innerHTML = `
      <div class="check btn" title="Toggle done">${ t.complete? '✔':'' }</div>
      <div class="title">${escapeHtml(t.title)}</div>
      <div class="ratio">${t.done}/${t.est}</div>
    `;
    const check = div.querySelector('.check');
    check.addEventListener('click', ()=>{
      if(!t.complete){
        if(t.done<t.est){ t.done++; }
        if(t.done>=t.est){ t.complete=true; }
      }else{
        t.complete=false; t.done=Math.max(0,t.est-1);
      }
      refreshTasks();
      checkAllDone();
    });
    div.addEventListener('click', ()=>{ activeTaskIndex=i; updateSliderLabel(); });
    return div;
  }

  function refreshTasks(){
    taskList.innerHTML='';
    tasks.forEach((t,i)=> taskList.appendChild(taskItem(t,i)));
    updateSummary();
    updateSliderLabel();
  }
  function updateSummary(){
    const total = tasks.reduce((a,t)=>a+t.est,0);
    const done = tasks.reduce((a,t)=>a+t.done,0);
    totalCountEl.textContent = total;
    doneCountEl.textContent = done;
    const remainingPomos = Math.max(0,total-done);
    const minutes = remainingPomos*25;
    const now = new Date();
    const doneMs = now.getTime() + minutes*60000;
    const dt = new Date(doneMs);
    const hh = dt.getHours().toString().padStart(2,'0');
    const mm = dt.getMinutes().toString().padStart(2,'0');
    finishAtEl.textContent = `${hh}:${mm}`;
    hoursEl.textContent = `(${(minutes/60).toFixed(1)}h)`;
  }
  function updateSliderLabel(){
    const idx = (activeTaskIndex>=0 && tasks[activeTaskIndex])?activeTaskIndex:0;
    currentIndexEl.textContent = (idx+1)||1;
  }
  function scrollToTasks(){ document.querySelector('.tasks').scrollIntoView({behavior:'smooth',block:'start'}); }

  function checkAllDone(){
    if(tasks.length && tasks.every(t=>t.complete)){
      // toast
      toast.classList.add('show');
      splashConfetti();
    }
  }

  clearTasks.addEventListener('click', ()=>{
    tasks = [];
    refreshTasks();
    toast.classList.remove('show');
  });

  // Modes click
  document.querySelectorAll('.mode').forEach(m=>{
    m.addEventListener('click', function(){
      switchMode(this.dataset.mode);
    });
  });

  startBtn.addEventListener('click', startPause);

  // Confetti
  function splashConfetti(){
    confettiBox.classList.add('active');
    confettiBox.innerHTML='';
    const colors=['#ffd166','#06d6a0','#118ab2','#ef476f','#ffffff'];
    for(let i=0;i<120;i++){
      const s = document.createElement('span');
      s.style.left = Math.random()*100+'%';
      s.style.background = colors[Math.floor(Math.random()*colors.length)];
      s.style.opacity = .9;
      s.style.transform = `translateY(-120vh) rotate(${Math.random()*360}deg)`;
      s.style.animationDelay = (Math.random()*0.2)+'s';
      s.style.width = (6+Math.random()*8)+'px';
      s.style.height = (8+Math.random()*10)+'px';
      confettiBox.appendChild(s);
    }
    setTimeout(()=>{ confettiBox.classList.remove('active'); },2400);
  }

  // Util
  function escapeHtml(str){ return str.replace(/[&<>"']/g, m=>({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[m])); }

  // Initial render
  setTheme();
  renderTime();
  refreshTasks();

  // Demo seed to match screenshots: create one example task after load
  setTimeout(()=>{
    tasks.push({id:1,title:'123',est:1,done:0,complete:false});
    refreshTasks();
  }, 200);

})();
</script>
</body>
