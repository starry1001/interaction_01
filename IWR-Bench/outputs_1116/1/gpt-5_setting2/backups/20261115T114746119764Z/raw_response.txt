<script>
/* 2048 core */
(function(){
  const size = 4;
  const tilesEl = document.getElementById('tiles');
  const boardEl = document.getElementById('board');
  const scoreEl = document.getElementById('score');
  const bestEl = document.getElementById('best');
  const addScoreEl = document.getElementById('addScore');
  const newGameBtn = document.getElementById('newGameBtn');

  const GAP = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--gap')) || 15;
  const TILE = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--size')) || 110;

  let grid = [];
  let score = 0;
  let best = parseInt(localStorage.getItem('best-2048') || '0', 10);

  bestEl.textContent = best;

  function reset(){
    grid = Array(size*size).fill(0);
    score = 0;
    scoreEl.textContent = '0';
    spawn(); spawn();
    render(true);
  }

  function index(r,c){ return r*size + c; }

  function randomEmpty(){
    const empty = [];
    for(let i=0;i<grid.length;i++) if(grid[i]===0) empty.push(i);
    if(!empty.length) return -1;
    return empty[Math.floor(Math.random()*empty.length)];
  }

  function spawn(){
    const i = randomEmpty();
    if(i<0) return false;
    grid[i] = Math.random() < 0.9 ? 2 : 4;
    return true;
  }

  function get(r,c){ return grid[index(r,c)]; }
  function set(r,c,val){ grid[index(r,c)] = val; }

  function move(dir){
    // dir: 0 left,1 up,2 right,3 down
    let moved = false;
    let gained = 0;
    const traverse = [];
    if(dir===0){ // left
      for(let r=0;r<size;r++) traverse.push({r,cStart:0,step:1});
    }else if(dir===2){ // right
      for(let r=0;r<size;r++) traverse.push({r,cStart:size-1,step:-1});
    }else if(dir===1){ // up
      for(let c=0;c<size;c++) traverse.push({c,rStart:0,step:1,vertical:true});
    }else if(dir===3){ // down
      for(let c=0;c<size;c++) traverse.push({c,rStart:size-1,step:-1,vertical:true});
    }
    const before = grid.slice().toString();

    for(const t of traverse){
      const line = [];
      if(t.vertical){
        for(let r=t.rStart, i=0; i<size; i++, r+=t.step) line.push(get(r,t.c));
      }else{
        for(let c=t.cStart, i=0; i<size; i++, c+=t.step) line.push(get(t.r,c));
      }
      const compact = line.filter(v=>v!==0);
      // merge
      const merged = [];
      for(let i=0;i<compact.length;i++){
        if(compact[i]===compact[i+1]){
          merged.push(compact[i]*2);
          gained += compact[i]*2;
          i++;
        }else merged.push(compact[i]);
      }
      while(merged.length<size) merged.push(0);
      // write back
      if(t.vertical){
        for(let r=t.rStart, i=0; i<size; i++, r+=t.step) set(r,t.c,merged[i]);
      }else{
        for(let c=t.cStart, i=0; i<size; i++, c+=t.step) set(t.r,c,merged[i]);
      }
    }

    const after = grid.slice().toString();
    moved = before !== after;
    if(moved){
      score += gained;
      if(gained>0) showAddScore(gained);
      updateScore();
      spawn();
      render();
      checkEnd();
    }
  }

  function updateScore(){
    scoreEl.textContent = score;
    if(score>best){
      best = score;
      localStorage.setItem('best-2048', best);
      bestEl.textContent = best;
    }
  }

  function showAddScore(val){
    addScoreEl.textContent = '+' + val;
    addScoreEl.classList.remove('show');
    void addScoreEl.offsetWidth; // reflow
    addScoreEl.classList.add('show');
    setTimeout(()=>addScoreEl.classList.remove('show'), 650);
  }

  function tilePos(r,c){
    return {
      left: c*(TILE+GAP),
      top:  r*(TILE+GAP)
    };
  }

  function render(initial=false){
    tilesEl.innerHTML = '';
    for(let r=0; r<size; r++){
      for(let c=0; c<size; c++){
        const v = get(r,c);
        if(!v) continue;
        const d = document.createElement('div');
        d.className = 'tile v-'+v + (initial ? ' new':'');
        const {left,top} = tilePos(r,c);
        d.style.left = left+'px';
        d.style.top = top+'px';
        d.textContent = v;
        // shrink font for larger numbers
        if(v>=1024) d.style.fontSize = '30px';
        else if(v>=128) d.style.fontSize = '36px';
        tilesEl.appendChild(d);
      }
    }
  }

  function checkEnd(){
    // any empty?
    if(grid.includes(0)) return;
    // any mergeable neighbors?
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        const v = get(r,c);
        if((c+1<size && get(r,c+1)===v) || (r+1<size && get(r+1,c)===v)) return;
      }
    }
    // game over banner (lightweight)
    const over = document.createElement('div');
    over.style.position='absolute';
    over.style.inset='0';
    over.style.background='rgba(238,228,218,.75)';
    over.style.borderRadius='6px';
    over.style.display='flex';
    over.style.alignItems='center';
    over.style.justifyContent='center';
    over.style.flexDirection='column';
    over.innerHTML = '<div style="font-size:48px;font-weight:800;color:#776e65;margin-bottom:8px">Game Over</div><button class="new-btn btn">Try again</button>';
    over.querySelector('button').addEventListener('click', ()=>{over.remove(); reset();});
    boardEl.appendChild(over);
  }

  // controls
  window.addEventListener('keydown', e=>{
    const map = {ArrowLeft:0, ArrowUp:1, ArrowRight:2, ArrowDown:3, a:0, w:1, d:2, s:3, A:0,W:1,D:2,S:3};
    if(map.hasOwnProperty(e.key)){ e.preventDefault(); move(map[e.key]); }
  });

  // touch / swipe
  let touchStart=null;
  boardEl.addEventListener('touchstart', e=>{
    const t = e.changedTouches[0];
    touchStart={x:t.clientX,y:t.clientY};
  },{passive:true});
  boardEl.addEventListener('touchend', e=>{
    if(!touchStart) return;
    const t = e.changedTouches[0];
    const dx = t.clientX - touchStart.x;
    const dy = t.clientY - touchStart.y;
    const absX = Math.abs(dx), absY = Math.abs(dy);
    if(Math.max(absX,absY) < 20) return;
    if(absX > absY) move(dx>0?2:0);
    else move(dy>0?3:1);
    touchStart=null;
  });

  newGameBtn.addEventListener('click', reset);

  // Initialize with a board resembling screenshots to feel familiar
  function preset(){ // small guided opening
    grid = Array(size*size).fill(0);
    // fill a few tiles like screenshots
    set(0,3,2);
    set(1,1,4);
    set(2,0,2);
    set(2,1,8);
    set(3,2,8);
    set(3,3,2);
    score = 36; updateScore();
    render(true);
  }

  // Start
  preset();

})();
</script>
</body>
