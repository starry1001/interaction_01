<script>
(function(){
  const size = 4;
  let grid = createEmpty();
  let score = 0;
  let best = Number(localStorage.getItem('best-2048') || 0);
  let isGameOver = false;

  const tilesEl = document.getElementById('tiles');
  const scoreEl = document.getElementById('score');
  const bestEl = document.getElementById('best');
  const overlay = document.getElementById('overlay');
  const overlayTitle = document.getElementById('overlayTitle');

  bestEl.textContent = best;

  function createEmpty(){
    return Array.from({length:size}, ()=> Array(size).fill(0));
  }

  function randomEmptyCell(){
    const empty = [];
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        if(grid[r][c] === 0) empty.push([r,c]);
      }
    }
    if(empty.length === 0) return null;
    return empty[Math.floor(Math.random()*empty.length)];
  }

  function addRandomTile(){
    const cell = randomEmptyCell();
    if(!cell) return;
    const [r,c] = cell;
    grid[r][c] = Math.random() < 0.9 ? 2 : 4;
  }

  function reset(){
    grid = createEmpty();
    score = 0;
    isGameOver = false;
    overlay.classList.remove('active');
    addRandomTile();
    addRandomTile();
    render(true);
  }

  function render(isReset=false){
    tilesEl.innerHTML = '';
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        const val = grid[r][c];
        const tile = document.createElement('div');
        tile.className = 'tile ' + (val? ('tile-' + val) : 'empty') + (isReset && val ? ' new' : '');
        tile.textContent = val || '';
        tile.style.gridRowStart = r+1;
        tile.style.gridColumnStart = c+1;
        tilesEl.appendChild(tile);
      }
    }
    scoreEl.textContent = score;
    if(score > best){
      best = score;
      bestEl.textContent = best;
      localStorage.setItem('best-2048', best);
    }
  }

  function move(dir){
    if(isGameOver) return;
    let moved=false;
    let gained=0;

    const traverse = (callback) => {
      for(let i=0; i<size; i++){
        let line = [];
        for(let j=0; j<size; j++){
          const [r,c] = callback(i,j);
          line.push(grid[r][c]);
        }
        let {newLine, mergedScore, didMove} = compressAndMerge(line);
        gained += mergedScore;
        if(didMove) moved=true;
        for(let j=0; j<size; j++){
          const [r,c] = callback(i,j);
          grid[r][c] = newLine[j];
        }
      }
    }

    if(dir==='left'){
      traverse((i,j)=>[i,j]);
    } else if(dir==='right'){
      traverse((i,j)=>[i, size-1-j]);
    } else if(dir==='up'){
      traverse((i,j)=>[j,i]);
    } else if(dir==='down'){
      traverse((i,j)=>[size-1-j,i]);
    }

    if(moved){
      score += gained;
      addRandomTile();
      render();
      if(checkGameOver()){
        isGameOver = true;
        overlayTitle.textContent = 'Game Over';
        overlay.classList.add('active');
      }
    }
  }

  function compressAndMerge(arr){
    const filtered = arr.filter(v=>v!==0);
    let mergedScore=0;
    let newLine=[];
    let didMove=false;
    for(let i=0; i<filtered.length; i++){
      if(filtered[i] === filtered[i+1]){
        const merged = filtered[i]*2;
        mergedScore += merged;
        newLine.push(merged);
        i++;
        didMove=true;
      }else{
        newLine.push(filtered[i]);
      }
    }
    while(newLine.length < size){ newLine.push(0); }
    // detect movement vs original
    if(!didMove){
      for(let i=0;i<size;i++){
        if(arr[i] !== newLine[i]){ didMove = true; break; }
      }
    }
    return {newLine, mergedScore, didMove};
  }

  function checkGameOver(){
    // any empty cell
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        if(grid[r][c]===0) return false;
      }
    }
    // any mergeable neighbors
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        const v=grid[r][c];
        if((r+1<size && grid[r+1][c]===v) || (c+1<size && grid[r][c+1]===v)) return false;
      }
    }
    return true;
  }

  // Controls
  document.addEventListener('keydown', (e)=>{
    const k = e.key.toLowerCase();
    if(['arrowleft','a'].includes(k)){ e.preventDefault(); move('left'); }
    if(['arrowright','d'].includes(k)){ e.preventDefault(); move('right'); }
    if(['arrowup','w'].includes(k)){ e.preventDefault(); move('up'); }
    if(['arrowdown','s'].includes(k)){ e.preventDefault(); move('down'); }
  });

  // Touch / swipe support
  let startX=0,startY=0;
  const boardEl=document.getElementById('board');
  boardEl.addEventListener('touchstart',(e)=>{
    const t=e.touches[0];
    startX=t.clientX; startY=t.clientY;
  },{passive:true});
  boardEl.addEventListener('touchend',(e)=>{
    const t=e.changedTouches[0];
    const dx=t.clientX-startX;
    const dy=t.clientY-startY;
    if(Math.abs(dx) < 20 && Math.abs(dy) < 20) return;
    if(Math.abs(dx) > Math.abs(dy)){
      if(dx>0) move('right'); else move('left');
    }else{
      if(dy>0) move('down'); else move('up');
    }
  });

  // Buttons
  document.getElementById('newGameBtn').addEventListener('click', reset);
  document.getElementById('restart').addEventListener('click', reset);
  document.getElementById('keepPlaying').addEventListener('click', ()=> overlay.classList.remove('active'));

  // Fake mode toggle (cosmetic only)
  const classicBtn = document.getElementById('modeClassic');
  const cupcakeBtn = document.getElementById('modeCupcakes');
  cupcakeBtn.addEventListener('click', ()=>{
    cupcakeBtn.classList.add('active');
    classicBtn.classList.remove('active');
    // cosmetic tweak: warm tint
    document.documentElement.style.setProperty('--board-bg', '#b49b87');
    document.documentElement.style.setProperty('--cell-bg', '#e7dbc7');
  });
  classicBtn.addEventListener('click', ()=>{
    classicBtn.classList.add('active');
    cupcakeBtn.classList.remove('active');
    document.documentElement.style.setProperty('--board-bg', '#bbada0');
    document.documentElement.style.setProperty('--cell-bg', '#cdc1b4');
  });

  // Initialize with a board that looks "alive"
  reset();

  // seed a few extra numbers to resemble the screenshots a bit (optional)
  function seedDemo(){
    grid = createEmpty();
    grid[0][1]=2; grid[0][3]=2;
    grid[1][2]=4;
    grid[2][0]=2; grid[2][2]=4; grid[2][3]=2;
    grid[3][1]=8; grid[3][3]=2;
    score=20;
    render(true);
  }
  // Uncomment the next line to start with a preset layout similar to screenshots:
  // seedDemo();

})();
</script>
