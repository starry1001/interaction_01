<script>
(function(){
  // 2048 Logic
  const size = 4;
  let board = [];
  let score = 0;
  const scoreEl = document.getElementById('score');
  const bestEl = document.getElementById('best');
  const boardEl = document.getElementById('board');
  const overlay = document.getElementById('overlay');
  const modalTitle = document.getElementById('modalTitle');
  const modalText = document.getElementById('modalText');
  const LS_BEST = 'best-2048-replica';

  function getBest(){
    return parseInt(localStorage.getItem(LS_BEST) || '0',10);
  }
  function setBest(v){ localStorage.setItem(LS_BEST, String(v)); bestEl.textContent = v; }

  function init(){
    board = Array.from({length:size}, _ => Array(size).fill(0));
    score = 0;
    scoreEl.textContent = '0';
    bestEl.textContent = getBest();
    addRandomTile();
    addRandomTile();
    render();
  }

  function addRandomTile(){
    const empties = [];
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        if(board[r][c]===0) empties.push({r,c});
      }
    }
    if(empties.length===0) return;
    const pick = empties[Math.floor(Math.random()*empties.length)];
    board[pick.r][pick.c] = Math.random()<0.9 ? 2 : 4;
  }

  function render(){
    // Build grid cells
    let html = '';
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        const v = board[r][c];
        const cls = v ? (' t'+v) : ' empty';
        const content = document.body.classList.contains('cupcakes') && v ? cupcakeLabel(v) : (v || '');
        html += `<div class="cell${cls.replace(' ', '')} ${cls.trim()}">${content}</div>`;
      }
    }
    boardEl.innerHTML = html;
  }

  function cupcakeLabel(v){
    // Little icing emoji + value
    return '🧁 ' + v;
  }

  function moveLeft(){
    let moved = false;
    for(let r=0;r<size;r++){
      const row = board[r].slice().filter(x=>x!==0);
      for(let c=0;c<row.length-1;c++){
        if(row[c]!==0 && row[c]===row[c+1]){
          row[c]*=2;
          score += row[c];
          row[c+1]=0;
          c++;
        }
      }
      const newRow = row.filter(x=>x!==0);
      while(newRow.length<size) newRow.push(0);
      if(newRow.some((v,i)=>v!==board[r][i])) moved = true;
      board[r] = newRow;
    }
    return moved;
  }

  function rotateClockwise(mat){
    const n = mat.length;
    const res = Array.from({length:n},()=>Array(n).fill(0));
    for(let i=0;i<n;i++){
      for(let j=0;j<n;j++){
        res[j][n-1-i] = mat[i][j];
      }
    }
    return res;
  }

  function move(dir){ // 'left','right','up','down'
    let moved=false;
    if(dir==='left'){
      moved = moveLeft();
    }else if(dir==='right'){
      // mirror horizontally: rotate twice and re-use left
      board = board.map(row => row.reverse());
      moved = moveLeft();
      board = board.map(row => row.reverse());
    }else if(dir==='up'){
      board = rotateClockwise(board); // right
      board = rotateClockwise(board); // down
      board = rotateClockwise(board); // left orientation
      moved = moveLeft();
      board = rotateClockwise(board); // back to up
    }else if(dir==='down'){
      board = rotateClockwise(board); // right
      moved = moveLeft();
      board = rotateClockwise(board); // down->back
      board = rotateClockwise(board);
      board = rotateClockwise(board);
    }
    if(moved){
      scoreEl.textContent = score;
      const best = getBest();
      if(score>best) setBest(score);
      addRandomTile();
      render();
      const state = checkState();
      if(state==='win'){
        openModal('You win!', 'Great job! You reached 2048. Keep going to increase your score, or start a new game.');
      }else if(state==='lose'){
        openModal('Game Over', 'No more moves are possible.');
      }
    }
  }

  function checkState(){
    // Win
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        if(board[r][c]===2048) return 'win';
      }
    }
    // Moves available?
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        const v = board[r][c];
        if(v===0) return 'playing';
        if(r+1<size && board[r+1][c]===v) return 'playing';
        if(c+1<size && board[r][c+1]===v) return 'playing';
      }
    }
    return 'lose';
  }

  // Controls
  window.addEventListener('keydown',(e)=>{
    const k = e.key.toLowerCase();
    if(['arrowleft','a'].includes(k)){ e.preventDefault(); move('left'); }
    else if(['arrowright','d'].includes(k)){ e.preventDefault(); move('right'); }
    else if(['arrowup','w'].includes(k)){ e.preventDefault(); move('up'); }
    else if(['arrowdown','s'].includes(k)){ e.preventDefault(); move('down'); }
  });

  // Touch controls
  let touchStart=null;
  boardEl.addEventListener('touchstart', (e)=>{
    const t = e.touches[0];
    touchStart = {x:t.clientX, y:t.clientY};
  }, {passive:true});
  boardEl.addEventListener('touchend', (e)=>{
    if(!touchStart) return;
    const t = e.changedTouches[0];
    const dx = t.clientX - touchStart.x;
    const dy = t.clientY - touchStart.y;
    if(Math.max(Math.abs(dx),Math.abs(dy))<24) return;
    if(Math.abs(dx)>Math.abs(dy)){
      if(dx>0) move('right'); else move('left');
    }else{
      if(dy>0) move('down'); else move('up');
    }
    touchStart=null;
  });

  // Buttons
  document.getElementById('newGameBtn').addEventListener('click', ()=>init());
  document.getElementById('tryAgainBtn').addEventListener('click', ()=>{ closeModal(); init(); });
  document.getElementById('closeModalBtn').addEventListener('click', closeModal);

  function openModal(title, text){
    modalTitle.textContent = title;
    modalText.textContent = text;
    overlay.classList.add('show');
  }
  function closeModal(){ overlay.classList.remove('show'); }

  // Fake "other games" interactions
  document.getElementById('otherGames').addEventListener('click', (e)=>{
    const card = e.target.closest('.card');
    if(!card) return;
    const name = card.getAttribute('data-game') || 'Game';
    openModal(name, 'This is a demo tile. In the original site this would open the ' + name + ' game.');
  });

  // Mode toggles
  const modeClassic = document.getElementById('modeClassic');
  const modeCupcakes = document.getElementById('modeCupcakes');
  modeClassic.addEventListener('click', ()=>{
    document.body.classList.remove('cupcakes');
    modeClassic.classList.add('active');
    modeCupcakes.classList.remove('active');
    render();
  });
  modeCupcakes.addEventListener('click', ()=>{
    document.body.classList.add('cupcakes');
    modeCupcakes.classList.add('active');
    modeClassic.classList.remove('active');
    render();
  });

  // Kick off
  init();
})();
</script>
