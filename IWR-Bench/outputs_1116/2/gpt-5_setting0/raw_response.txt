<script>
(function(){
  // Fake track data
  const TRACK = {
    title:"Sacrifice",
    artist:"Mariah the Scientist — HEARTS SOLD SEPARATELY",
    length: 79 // seconds
  };

  // Elements
  const overlay = document.getElementById('overlay');
  const miniOpen = document.getElementById('miniOpen');
  const ctaStart = document.getElementById('ctaStart');
  const overlayClose = document.getElementById('overlayClose');

  const topBar = document.getElementById('topBar');
  const bottomBar = document.getElementById('bottomBar');
  const miniProg = document.getElementById('miniProg');

  const timeCurrent = document.getElementById('timeCurrent');
  const timeLeft = document.getElementById('timeLeft');
  const durationLabel = document.getElementById('durationLabel');

  const playIcon = document.getElementById('playIcon');
  const headerPlayIcon = document.getElementById('headerPlayIcon');

  const playToggle = document.getElementById('playToggle');
  const playToggleHeader = document.getElementById('playToggleHeader');

  // State
  let playing = false;
  let t = 0;
  let timer = null;

  function fmt(seconds){
    const s = Math.max(0, Math.floor(seconds));
    return Math.floor(s/60)+':'+String(s%60).padStart(2,'0');
  }

  function updateUI(){
    const pct = (t/TRACK.length)*100;
    topBar.style.width = pct+'%';
    bottomBar.style.width = pct+'%';
    miniProg.style.width = pct+'%';

    timeCurrent.textContent = fmt(t);
    timeLeft.textContent = '-' + fmt(TRACK.length - t);
    durationLabel.textContent = '-' + fmt(TRACK.length - t);

    // Toggle icons
    if(playing){
      playIcon.src = "__PLACEHOLDER_ASSETS_BASE_DIR__/008.svg"; // pause
      headerPlayIcon.src = "__PLACEHOLDER_ASSETS_BASE_DIR__/008.svg";
    }else{
      playIcon.src = "__PLACEHOLDER_ASSETS_BASE_DIR__/011.svg"; // play
      headerPlayIcon.src = "__PLACEHOLDER_ASSETS_BASE_DIR__/011.svg";
    }
  }

  function startTimer(){
    if(timer) return;
    playing = true;
    timer = setInterval(()=>{
      t += 0.5;
      if(t >= TRACK.length){
        t = TRACK.length;
        stopTimer();
      }
      updateUI();
    }, 500);
    updateUI();
  }
  function stopTimer(){
    playing = false;
    clearInterval(timer);
    timer = null;
    updateUI();
  }

  // Open/Close overlay
  function openOverlay(){
    overlay.classList.add('show');
  }
  function closeOverlay(){
    overlay.classList.remove('show');
  }

  // Event bindings
  miniOpen.addEventListener('click', openOverlay);
  ctaStart.addEventListener('click', openOverlay);
  overlayClose.addEventListener('click', closeOverlay);

  playToggle.addEventListener('click', ()=>{
    playing ? stopTimer() : startTimer();
  });
  playToggleHeader.addEventListener('click', ()=>{
    playing ? stopTimer() : startTimer();
  });

  // Prev / Next simulate jump
  function prev(){
    t = Math.max(0, t-10);
    updateUI();
  }
  function next(){
    t = Math.min(TRACK.length, t+10);
    updateUI();
  }
  document.getElementById('prevBtn').addEventListener('click', prev);
  document.getElementById('prevBtn2').addEventListener('click', prev);
  document.getElementById('nextBtn').addEventListener('click', next);
  document.getElementById('nextBtn2').addEventListener('click', next);

  // Start in paused preview state but with tiny progress for realism
  t = 7;
  updateUI();
})();
</script>
