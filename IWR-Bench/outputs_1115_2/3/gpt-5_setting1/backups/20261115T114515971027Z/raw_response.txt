<script>
(function(){
  const timeEl = document.getElementById('time');
  const toggleBtn = document.getElementById('toggleBtn');
  const quickPlay = document.getElementById('quickPlay');
  const sessionMessage = document.getElementById('sessionMessage');
  const sessionCounter = document.getElementById('sessionCounter');
  const modes = document.querySelectorAll('.mode-tab');
  const addTaskBtn = document.getElementById('addTaskBtn');
  const taskForm = document.getElementById('taskForm');
  const taskInput = document.getElementById('taskInput');
  const estInput = document.getElementById('estInput');
  const downEst = document.getElementById('downEst');
  const upEst = document.getElementById('upEst');
  const cancelTask = document.getElementById('cancelTask');
  const saveTask = document.getElementById('saveTask');
  const taskList = document.getElementById('taskList');
  const summary = document.getElementById('summary');
  const toast = document.getElementById('toast');
  const clearTasks = document.getElementById('clearTasks');
  const confetti = document.getElementById('confetti');

  let mode = 'pomo'; // pomo | short | long
  const durations = {pomo:25*60, short:5*60, long:15*60};
  let sessionIndex = 1;
  let remaining = durations[mode];
  let ticking = null;
  let running = false;

  const tasks = []; // {id,title,est,done,progress}
  let taskId = 1;

  function fmt(sec){
    const m = Math.floor(sec/60).toString().padStart(2,'0');
    const s = Math.floor(sec%60).toString().padStart(2,'0');
    return m+':'+s;
  }

  function setMode(newMode){
    mode = newMode;
    document.body.classList.remove('mode-pomo','mode-short','mode-long');
    document.body.classList.add('mode-'+(newMode==='pomo'?'pomo': newMode));
    modes.forEach(m => m.classList.toggle('active', m.dataset.mode === newMode));
    remaining = durations[newMode];
    timeEl.textContent = fmt(remaining);
    if(newMode === 'pomo'){
      sessionMessage.textContent = 'Time to focus!';
    }else{
      sessionMessage.textContent = 'Time for a break!';
    }
    toggleBtn.textContent = 'START';
    running = false;
    if(ticking){ clearInterval(ticking); ticking = null; }
  }

  function start(){
    if(running) return;
    running = true;
    toggleBtn.textContent = 'PAUSE';
    ticking = setInterval(()=>{
      remaining--;
      timeEl.textContent = fmt(remaining);
      if(remaining<=0){
        clearInterval(ticking); ticking=null; running=false;
        toggleBtn.textContent='START';
        // auto mark progress of first task
        if(tasks.length){
          const t = tasks[0];
          t.progress = Math.min(t.est, t.progress+1);
          if(t.progress>=t.est){ t.done=true; }
          renderTasks();
          if(allDone()){
            showToast();
            launchConfetti();
          }
        }
        // reset for same mode
        remaining = durations[mode];
        timeEl.textContent = fmt(remaining);
      }
    },1000);
  }
  function pause(){
    if(!running) return;
    running = false;
    toggleBtn.textContent = 'START';
    if(ticking){ clearInterval(ticking); ticking=null; }
  }

  toggleBtn.addEventListener('click', ()=> running?pause():start() );
  quickPlay.addEventListener('click', ()=> running?pause():start() );

  modes.forEach(b=> b.addEventListener('click', ()=> setMode(b.dataset.mode)));

  // Task form logic
  addTaskBtn.addEventListener('click', ()=>{
    taskForm.classList.remove('hidden');
    taskInput.focus();
  });
  cancelTask.addEventListener('click', ()=> {
    taskForm.classList.add('hidden');
    taskInput.value='';
    estInput.value=1;
  });
  downEst.addEventListener('click', ()=> {
    estInput.value = Math.max(1, parseInt(estInput.value||1)-1);
  });
  upEst.addEventListener('click', ()=> {
    estInput.value = Math.max(1, parseInt(estInput.value||1)+1);
  });

  saveTask.addEventListener('click', ()=>{
    const title = taskInput.value.trim() || 'Untitled';
    const est = Math.max(1, parseInt(estInput.value||1));
    const t = {id:taskId++, title, est, done:false, progress:0};
    tasks.push(t);
    taskInput.value=''; estInput.value=1;
    taskForm.classList.add('hidden');
    renderTasks();
  });

  function renderTasks(){
    taskList.innerHTML='';
    tasks.forEach(t=>{
      const li = document.createElement('div');
      li.className = 'task'+(t.done?' done':'');
      li.innerHTML = `
        <div class="check btn" data-id="${t.id}">
          <img src="__PLACEHOLDER_ASSETS_BASE_DIR__/005.png" width="14" height="14" alt="done">
        </div>
        <div class="title">${escapeHtml(t.title)}</div>
        <div class="count">${t.progress}/${t.est}</div>
        <div class="more btn"><img src="__PLACEHOLDER_ASSETS_BASE_DIR__/003.png" width="14" height="14" alt="more"></div>
      `;
      taskList.appendChild(li);
    });

    // events
    taskList.querySelectorAll('.check').forEach(btn=>{
      btn.addEventListener('click', ()=>{
        const id = parseInt(btn.getAttribute('data-id'));
        const t = tasks.find(x=>x.id===id);
        if(!t) return;
        if(!t.done){
          t.progress = t.est;
          t.done = true;
        }else{
          t.progress = 0;
          t.done = false;
        }
        renderTasks();
        if(allDone() && tasks.length){ showToast(); launchConfetti(); }
      });
    });

    // header current task
    if(tasks.length){
      const active = tasks.find(t=>!t.done) || tasks[0];
      document.getElementById('sessionCounter').textContent = '#'+sessionIndex;
      document.getElementById('sessionMessage').previousElementSibling?.remove;
      sessionMessage.parentNode.querySelector ? null : null;
      document.getElementById('sessionCounter').nextElementSibling;
      document.getElementById('sessionCounter');
      document.getElementById('sessionCounter').dataset.title = active.title;
      // show active title below timer, similar to screenshots
      sessionMessage.textContent = (mode==='pomo')?'Time to focus!':'Time for a break!';
      // also show small label of current title on top area
      // We'll put the title right beneath counter via extra line
      sessionCounter.textContent = '#1';
      const currentLabel = document.getElementById('currentTask');
      if(currentLabel){ currentLabel.remove(); }
      const label = document.createElement('div');
      label.id='currentTask';
      label.style.textAlign='center';
      label.style.marginTop='6px';
      label.style.opacity='0.9';
      label.textContent = active.title;
      sessionMessage.insertAdjacentElement('afterend', label);
    }else{
      const currentLabel = document.getElementById('currentTask');
      if(currentLabel){ currentLabel.remove(); }
    }

    // summary
    if(tasks.length){
      summary.classList.remove('hidden');
      const totalEst = tasks.reduce((a,b)=>a+b.est,0);
      const doneEst = tasks.reduce((a,b)=>a+b.progress,0);
      const remainEst = totalEst - doneEst;
      const mins = remainEst * (mode==='pomo'?25:(mode==='short'?5:15));
      const finish = new Date(Date.now() + mins*60000);
      const hh = finish.getHours().toString().padStart(2,'0');
      const mm = finish.getMinutes().toString().padStart(2,'0');
      summary.textContent = `Pomos: ${doneEst}/${totalEst}   Finish At: ${hh}:${mm} (${(mins/60).toFixed(1)}h)`;
    }else{
      summary.classList.add('hidden');
    }
    // show Add-task box below list always
    const add = document.getElementById('addTaskBtn');
    add.parentNode.insertBefore(add, summary);
  }

  function allDone(){
    return tasks.length && tasks.every(t=>t.done);
  }

  function showToast(){
    toast.classList.add('show');
    setTimeout(()=>toast.classList.remove('show'), 4000);
  }
  clearTasks.addEventListener('click', ()=>{
    tasks.splice(0,tasks.length);
    renderTasks();
    toast.classList.remove('show');
  });

  function launchConfetti(){
    confetti.classList.add('show');
    confetti.innerHTML='';
    const colors = ['#ffcc00','#ff5a5f','#2ec4b6','#a78bfa','#34d399','#f472b6','#60a5fa'];
    for(let i=0;i<120;i++){
      const s = document.createElement('span');
      s.style.left = Math.random()*100+'vw';
      s.style.background = colors[Math.floor(Math.random()*colors.length)];
      s.style.animationDelay = (Math.random()*0.8)+'s';
      s.style.transform = 'rotate('+ (Math.random()*360)+'deg)';
      confetti.appendChild(s);
    }
    setTimeout(()=>confetti.classList.remove('show'), 2500);
  }

  // helpers
  function escapeHtml(str){
    return str.replace(/[&<>"']/g,function(m){
      return ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]);
    });
  }

  // Init
  setMode('pomo');
  timeEl.textContent = fmt(remaining);
})();
</script>
</body>
</html>
