<script>
(function(){
  const modes = {
    pomodoro:{ min:25, bg:'mode-pomodoro', focusText:'Time to focus!' },
    short:{ min:5, bg:'mode-short', focusText:'Time for a break!' },
    long:{ min:15, bg:'mode-long', focusText:'Time for a break!' }
  };
  let currentMode = 'pomodoro';
  let secondsLeft = modes[currentMode].min * 60;
  let timer = null;
  let running = false;

  const timeEl = document.getElementById('time');
  const startBtn = document.getElementById('startBtn');
  const tabs = document.querySelectorAll('.tab');
  const sessionNote = document.getElementById('currentTaskNote');

  const addTaskBox = document.getElementById('addTaskBox');
  const taskForm = document.getElementById('taskForm');
  const cancelTaskBtn = document.getElementById('cancelTask');
  const saveTaskBtn = document.getElementById('saveTask');
  const taskTitleEl = document.getElementById('taskTitle');
  const taskEstEl = document.getElementById('taskEst');
  const upBtn = document.getElementById('upBtn');
  const downBtn = document.getElementById('downBtn');
  const taskList = document.getElementById('taskList');

  const summary = document.getElementById('summary');
  const pomoDoneEl = document.getElementById('pomoDone');
  const pomoTotalEl = document.getElementById('pomoTotal');
  const finishAtEl = document.getElementById('finishAt');
  const finishHoursEl = document.getElementById('finishHours');

  const toast = document.getElementById('toast');
  const clearTasksBtn = document.getElementById('clearTasks');

  const confetti = document.getElementById('confetti');

  let tasks = [];

  function two(n){ return n<10 ? '0'+n : ''+n; }

  function renderTime(){
    const m = Math.floor(secondsLeft/60);
    const s = secondsLeft%60;
    timeEl.textContent = two(m) + ':' + two(s);
  }

  function setMode(mode){
    currentMode = mode;
    document.body.classList.remove('mode-pomodoro','mode-short','mode-long');
    document.body.classList.add(modes[mode].bg);
    tabs.forEach(t=>t.classList.toggle('active', t.dataset.mode===mode));
    secondsLeft = modes[mode].min*60;
    running = false;
    clearInterval(timer); timer=null;
    startBtn.textContent = 'START';
    sessionNote.textContent = modes[mode].focusText;
    renderTime();
  }

  function tick(){
    if(secondsLeft>0){ secondsLeft--; renderTime(); }
    if(secondsLeft<=0){
      clearInterval(timer); running=false;
      startBtn.textContent='START';
    }
  }

  startBtn.addEventListener('click', ()=>{
    if(!running){
      running=true;
      startBtn.innerHTML='PAUSE <img class="play-ico" src="__PLACEHOLDER_ASSETS_BASE_DIR__/008.png" width="16" height="16" alt="play">';
      timer=setInterval(tick,1000);
    }else{
      running=false;
      clearInterval(timer);
      startBtn.textContent='START';
    }
  });

  tabs.forEach(tab=>{
    tab.addEventListener('click', ()=> setMode(tab.dataset.mode));
  });

  // Task form controls
  addTaskBox.addEventListener('click', ()=> {
    addTaskBox.style.display='none';
    taskForm.style.display='block';
    taskTitleEl.focus();
  });
  cancelTaskBtn.addEventListener('click', ()=> {
    taskForm.style.display='none';
    addTaskBox.style.display='flex';
    taskTitleEl.value='';
    taskEstEl.value='1';
  });
  upBtn.addEventListener('click', ()=> taskEstEl.value = (+taskEstEl.value||1)+1 );
  downBtn.addEventListener('click', ()=> taskEstEl.value = Math.max(1,(+taskEstEl.value||1)-1) );

  function updateSummary(){
    const total = tasks.reduce((a,t)=>a + (t.est||1),0);
    const done = tasks.filter(t=>t.done).length===tasks.length && tasks.length>0 ? total : tasks.reduce((a,t)=>a+(t.done?t.estDone:0),0);
    // For simple demo we just show completed pomodoros 0 and total
    pomoDoneEl.textContent = 0;
    pomoTotalEl.textContent = total;
    if(total>0){
      summary.style.display='flex';
      // estimate finish time: now + total*25 minutes
      const addMin = total*25;
      const end = new Date(Date.now()+addMin*60*1000);
      finishAtEl.textContent = two(end.getHours())+':'+two(end.getMinutes());
      finishHoursEl.textContent = (Math.round((addMin/60)*10)/10)+'h';
    } else {
      summary.style.display='none';
    }
  }

  function renderTasks(){
    taskList.innerHTML='';
    tasks.forEach((t,i)=>{
      const row = document.createElement('div');
      row.className='task-item';
      const left = document.createElement('div');
      left.className='task-left';

      const check = document.createElement('div');
      check.className='check btn';
      const ck = document.createElement('img');
      ck.src='__PLACEHOLDER_ASSETS_BASE_DIR__/016.png';
      ck.width=18; ck.height=18; ck.alt='check';
      check.appendChild(ck);
      check.addEventListener('click', ()=>{
        t.done = !t.done;
        if(t.done){ row.style.opacity=.6; row.querySelector('.task-title').style.textDecoration='line-through'; }
        else { row.style.opacity=1; row.querySelector('.task-title').style.textDecoration='none'; }
        checkAllDone();
      });

      const title = document.createElement('div');
      title.className='task-title';
      title.textContent = t.title;

      left.appendChild(check);
      left.appendChild(title);

      const right = document.createElement('div');
      right.className='task-right';
      right.textContent = '0 / '+t.est;

      row.appendChild(left);
      row.appendChild(right);
      taskList.appendChild(row);
    });
    // update header under timer with first task if exists
    const note = document.getElementById('currentTaskNote');
    if(tasks.length>0){
      note.textContent = tasks[0].title;
    }else{
      note.textContent = modes[currentMode].focusText;
    }
    updateSummary();
  }

  function checkAllDone(){
    const allDone = tasks.length>0 && tasks.every(t=>t.done);
    if(allDone){
      toast.style.display='flex';
      fireConfetti();
    }else{
      toast.style.display='none';
    }
    renderTasks();
  }

  saveTaskBtn.addEventListener('click', ()=>{
    const title = taskTitleEl.value.trim();
    const est = Math.max(1, parseInt(taskEstEl.value||'1',10));
    if(!title) return;
    tasks.push({ title, est, estDone:0, done:false });
    taskTitleEl.value='';
    taskEstEl.value='1';
    taskForm.style.display='none';
    addTaskBox.style.display='flex';
    renderTasks();
  });

  clearTasksBtn.addEventListener('click', ()=>{
    tasks = [];
    renderTasks();
    toast.style.display='none';
  });

  // Confetti
  function fireConfetti(){
    confetti.style.display='block';
    confetti.innerHTML='';
    const colors=['#ff595e','#ffca3a','#8ac926','#1982c4','#6a4c93','#ffffff'];
    for(let i=0;i<80;i++){
      const p=document.createElement('div');
      p.className='piece';
      p.style.left = Math.random()*100 + 'vw';
      p.style.animationDelay = (Math.random()*400)+'ms';
      p.style.background = colors[Math.floor(Math.random()*colors.length)];
      confetti.appendChild(p);
    }
    setTimeout(()=>{ confetti.style.display='none'; },2100);
  }

  // initial render
  renderTime();
  renderTasks();
})();
</script>
</body>
</html>
