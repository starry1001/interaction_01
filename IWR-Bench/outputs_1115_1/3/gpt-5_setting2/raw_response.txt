<script>
(function(){
  const modes = {
    pomodoro:{minutes:25, theme:'theme-pomodoro', msg:'Time to focus!'},
    short:{minutes:5, theme:'theme-short', msg:'Time for a break!'},
    long:{minutes:15, theme:'theme-long', msg:'Time for a break!'},
  };

  let state = {
    mode:'pomodoro',
    remaining: modes.pomodoro.minutes*60,
    running:false,
    interval:null,
    tasks:[],
    nextId:1
  };

  // Elements
  const timeEl = document.getElementById('time');
  const startBtn = document.getElementById('startBtn');
  const messageEl = document.getElementById('message');
  const tabs = document.querySelectorAll('.tab');
  const tasksWrap = document.getElementById('tasks');
  const addTaskTrigger = document.getElementById('addTaskTrigger');
  const bottomPomos = document.getElementById('pomoProgress');
  const finishAt = document.getElementById('finishAt');
  const finishHours = document.getElementById('finishHours');
  const toast = document.getElementById('toast');

  function fmt(sec){
    const m = Math.floor(sec/60).toString().padStart(2,'0');
    const s = Math.floor(sec%60).toString().padStart(2,'0');
    return `${m}:${s}`;
  }

  function setMode(m){
    state.mode = m;
    state.remaining = modes[m].minutes*60;
    state.running = false;
    clearInterval(state.interval);
    startBtn.textContent='START';
    timeEl.textContent = fmt(state.remaining);
    document.body.className = modes[m].theme;
    messageEl.textContent = `#1 ${modes[m].msg}`;
    tabs.forEach(t=>t.classList.toggle('active', t.dataset.mode===m));
  }

  function tick(){
    if(state.remaining>0){
      state.remaining--;
      timeEl.textContent = fmt(state.remaining);
    }else{
      clearInterval(state.interval);
      state.running=false;
      startBtn.textContent='START';
    }
  }

  startBtn.addEventListener('click', function(){
    if(!state.running){
      state.running = true;
      startBtn.textContent='PAUSE';
      state.interval = setInterval(tick,1000);
    }else{
      state.running = false;
      startBtn.textContent='START';
      clearInterval(state.interval);
    }
  });

  tabs.forEach(tab=>{
    tab.addEventListener('click', () => setMode(tab.dataset.mode));
  });

  // Tasks handling
  function render(){
    // clear except final add box
    tasksWrap.innerHTML = '';
    // editor or items
    state.tasks.forEach(t=>{
      const item = document.createElement('div');
      item.className = 'task-item' + (t.done?' done':'');
      item.dataset.id = t.id;

      const left = document.createElement('div');
      left.className = 'task-left';
      const check = document.createElement('div');
      check.className = 'check btn';
      const img = document.createElement('img');
      img.src = document.getElementById('checkAsset').src;
      img.width=18; img.height=18; img.alt='check';
      check.appendChild(img);
      check.addEventListener('click',()=>{
        t.done = !t.done;
        // when done, mark pomodoros completed
        if(t.done){t.completed = t.est;}
        updateFooter();
        render();
        if(allDone() && state.tasks.length>0){
          showToast();
          fireConfetti();
        }
      });

      const title = document.createElement('div');
      title.innerHTML = `<div class="task-title">${escapeHTML(t.title||'Untitled')}</div>
                         <div class="task-meta">${t.completed}/${t.est}</div>`;

      left.appendChild(check);
      left.appendChild(title);

      const right = document.createElement('div');
      right.className='task-right';

      const more = document.createElement('div');
      more.className='more btn';
      const moreImg = document.createElement('img');
      moreImg.src='__PLACEHOLDER_ASSETS_BASE_DIR__/003.png';
      moreImg.width=16; moreImg.height=16; moreImg.alt='menu';
      more.appendChild(moreImg);

      right.appendChild(more);

      item.appendChild(left);
      item.appendChild(right);
      tasksWrap.appendChild(item);
    });

    // add editor if pending
    if(document.querySelector('.task-editor')==null && addEditorPending){
      tasksWrap.appendChild(makeEditor());
    }

    // add box
    const addBox = document.createElement('div');
    addBox.className='add-task-box btn';
    addBox.id='addTaskTrigger';
    addBox.textContent='+ Add Task';
    addBox.addEventListener('click', ()=>{
      if(document.querySelector('.task-editor')) return;
      addEditorPending = true;
      render();
    });
    tasksWrap.appendChild(addBox);

    updateFooter();
  }

  function updateFooter(){
    const total = state.tasks.reduce((s,t)=>s+t.est,0);
    const done = state.tasks.reduce((s,t)=>s+t.completed,0);
    bottomPomos.textContent = `${done}/${total}`;
    const minsRemaining = Math.max(total-done,0) * 25;
    const now = new Date();
    const finish = new Date(now.getTime()+minsRemaining*60000);
    const hh = finish.getHours().toString().padStart(2,'0');
    const mm = finish.getMinutes().toString().padStart(2,'0');
    finishAt.textContent = `${hh}:${mm}`;
    finishHours.textContent = `${(minsRemaining/60).toFixed(1)}h`;
  }

  function allDone(){
    return state.tasks.length>0 && state.tasks.every(t=>t.done);
  }

  function escapeHTML(str){
    return str.replace(/[&<>"']/g, s=>({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[s]));
  }

  let addEditorPending = false;

  function makeEditor(){
    const ed = document.createElement('div');
    ed.className='task-editor';

    ed.innerHTML = `
      <textarea placeholder="What are you working on?"></textarea>
      <div class="est-row">
        <span>Est Pomodoros</span>
        <span class="pill btn" data-act="dec">-</span>
        <strong id="estVal">1</strong>
        <span class="pill btn" data-act="inc">+</span>
      </div>
      <div class="muted-row">
        <span class="muted-link btn">+ Add Note</span>
        <span class="muted-link btn">+ Add Project</span>
      </div>
      <div class="editor-actions">
        <div class="btn btn-cancel">Cancel</div>
        <div class="btn btn-save">Save</div>
      </div>
    `;

    const textarea = ed.querySelector('textarea');
    const estVal = ed.querySelector('#estVal');
    ed.querySelectorAll('.pill').forEach(p=>{
      p.addEventListener('click', ()=>{
        let v = parseInt(estVal.textContent,10);
        if(p.dataset.act==='inc') v++;
        else v = Math.max(1,v-1);
        estVal.textContent = v;
      });
    });

    ed.querySelector('.btn-cancel').addEventListener('click', ()=>{
      addEditorPending=false;
      render();
    });

    ed.querySelector('.btn-save').addEventListener('click', ()=>{
      state.tasks.push({
        id: state.nextId++,
        title: textarea.value.trim() || 'Task',
        est: parseInt(estVal.textContent,10),
        completed:0,
        done:false
      });
      addEditorPending=false;
      render();
      // Scroll to tasks section
      setTimeout(()=>window.scrollTo({top:400,behavior:'smooth'}),0);
    });

    return ed;
  }

  addTaskTrigger.addEventListener('click', ()=>{
    if(document.querySelector('.task-editor')) return;
    addEditorPending = true;
    render();
  });

  // Toast
  function showToast(){
    toast.style.display='flex';
    clearTimeout(showToast._t);
    showToast._t = setTimeout(()=>{toast.style.display='none'}, 5000);
  }
  document.getElementById('clearTasksBtn').addEventListener('click', function(){
    state.tasks=[];
    render();
    toast.style.display='none';
    fireConfetti(); // confetti again as in sample
  });

  // Confetti
  const canvas = document.getElementById('confetti');
  const ctx = canvas.getContext('2d');
  let confettiPieces=[];
  function fireConfetti(){
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
    canvas.style.display='block';
    confettiPieces = [];
    const colors = ['#ffd166','#ef476f','#06d6a0','#118ab2','#8338ec','#ff9f1c'];
    for(let i=0;i<140;i++){
      confettiPieces.push({
        x: Math.random()*canvas.width,
        y: -10-Math.random()*canvas.height*0.5,
        r: 4+Math.random()*6,
        c: colors[Math.floor(Math.random()*colors.length)],
        s: 2+Math.random()*3,
        a: Math.random()*Math.PI
      });
    }
    let t = 0;
    function anim(){
      ctx.clearRect(0,0,canvas.width,canvas.height);
      confettiPieces.forEach(p=>{
        p.y += p.s;
        p.x += Math.sin(p.a+t/20)*0.8;
        ctx.fillStyle=p.c;
        ctx.save();
        ctx.translate(p.x,p.y);
        ctx.rotate(p.a);
        ctx.fillRect(-p.r/2,-p.r/2,p.r,p.r);
        ctx.restore();
      });
      t++;
      if(t<180){
        requestAnimationFrame(anim);
      }else{
        canvas.style.display='none';
      }
    }
    anim();
  }

  // Initialize
  setMode('pomodoro');
  render();

  // Accessibility: clicking header icons does nothing but are interactive
  document.querySelectorAll('.hdr-actions .btn, .brand').forEach(b=>b.addEventListener('click',()=>{}));

  // Resize canvas with window
  window.addEventListener('resize', ()=>{
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
  });
})();
</script>
</body>
</html>
