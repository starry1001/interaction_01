<script>
/* Mock track data and simple player state */
const track = {
  title: "Sacrifice",
  artist: "Mariah the Scientist",
  album: "HEARTS SOLD SEPARATELY",
  duration: 115 // seconds (for preview)
};

let playing = false;
let cur = 0; // seconds
let timer = null;

const fmt = s=>{
  s = Math.max(0, Math.floor(s));
  const m = Math.floor(s/60);
  const ss = String(s%60).padStart(2,'0');
  return `${m}:${ss}`;
};
const updateUI = ()=>{
  // overlay
  document.getElementById('curTime').textContent = fmt(cur);
  document.getElementById('durTime').textContent = `-${fmt(track.duration - cur)}`;
  document.getElementById('seek').value = cur;

  // mini bar
  const pct = (cur/track.duration)*100;
  document.getElementById('miniFill').style.width = pct + '%';

  // toggles
  document.getElementById('iconPlay').classList.toggle('hidden', playing);
  document.getElementById('iconPause').classList.toggle('hidden', !playing);
  document.getElementById('npPlay').classList.toggle('hidden', playing);
  document.getElementById('npPause').classList.toggle('hidden', !playing);
};
const stopTimer = ()=>{ if(timer){ clearInterval(timer); timer=null; } };
const startTimer = ()=>{
  stopTimer();
  timer = setInterval(()=>{
    if(!playing) return;
    cur += 0.5;
    if(cur>=track.duration){
      cur = track.duration;
      playing = false;
      stopTimer();
    }
    updateUI();
  }, 500);
};

document.getElementById('btnPlayPause').addEventListener('click', ()=>{
  playing = !playing;
  startTimer();
  updateUI();
});
document.getElementById('npPlayPause').addEventListener('click', ()=>{
  playing = !playing;
  startTimer();
  updateUI();
});
document.getElementById('btnPrev').addEventListener('click', ()=>{ cur=0; updateUI(); });
document.getElementById('btnNext').addEventListener('click', ()=>{ cur=track.duration; updateUI(); });
document.getElementById('npPrev').addEventListener('click', ()=>{ cur=0; updateUI(); });
document.getElementById('npNext').addEventListener('click', ()=>{ cur=track.duration; updateUI(); });
document.getElementById('npBack10').addEventListener('click', ()=>{ cur=Math.max(0,cur-10); updateUI(); });
document.getElementById('npFwd10').addEventListener('click', ()=>{ cur=Math.min(track.duration,cur+10); updateUI(); });

document.getElementById('seek').addEventListener('input', e=>{
  cur = +e.target.value;
  updateUI();
});
document.getElementById('openOverlay').addEventListener('click', ()=>{
  document.getElementById('overlay').classList.add('show');
});
document.getElementById('closeOverlay').addEventListener('click', ()=>{
  document.getElementById('overlay').classList.remove('show');
});

let fav=false;
document.getElementById('btnFav').addEventListener('click', ()=>{
  fav = !fav;
  const ic = document.getElementById('favIcon');
  ic.style.filter = fav ? 'invert(1) sepia(1) saturate(5) hue-rotate(320deg) brightness(1.3)' : 'invert(1) brightness(2)';
});

updateUI();
</script>
