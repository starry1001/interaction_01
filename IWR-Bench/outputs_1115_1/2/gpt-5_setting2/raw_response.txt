<script>
  // Fake content by section for hero
  const heroData = {
    home: {
      title: "Discover new music every day.",
      desc: "Get playlists and albums inspired by the artists and genres you’re listening to.",
      start:"#e11d48", end:"#ff3e70"
    },
    new: {
      title:"New releases.",
      desc:"Fresh tracks added daily—start listening now.",
      start:"#f43f5e", end:"#fb7185"
    },
    radio: {
      title:"Radio you’ll love.",
      desc:"Stations and live shows crafted by music experts.",
      start:"#c026d3", end:"#f43f5e"
    }
  };

  const heroTitle = document.querySelector('.hero-title');
  const heroDesc  = document.querySelector('.hero-desc');
  const hero = document.querySelector('.hero');
  document.querySelectorAll('.nav-item').forEach(item=>{
    item.addEventListener('click', ()=>{
      document.querySelectorAll('.nav-item').forEach(i=>i.classList.remove('active'));
      item.classList.add('active');
      const key = item.getAttribute('data-section');
      const d = heroData[key];
      heroTitle.textContent = d.title;
      heroDesc.textContent = d.desc;
      document.documentElement.style.setProperty('--hero-start', d.start);
      document.documentElement.style.setProperty('--hero-end', d.end);
    });
  });

  // Player overlay logic
  const player = document.getElementById('player');
  const openers = document.querySelectorAll('.open-player');
  const closeBtn = document.getElementById('closePlayer');
  openers.forEach(el=>el.addEventListener('click', ()=>player.classList.add('show')));
  closeBtn.addEventListener('click', ()=>player.classList.remove('show'));

  // Simulated playback
  const playPause = document.getElementById('playPause');
  const playPauseIcon = document.getElementById('playPauseIcon');
  const progressFill = document.getElementById('progressFill');
  const elapsedEl = document.getElementById('elapsed');
  const durationEl = document.getElementById('duration');
  const progressBar = document.getElementById('progressBar');

  let isPlaying = false;
  const total = 95; // fake seconds
  let t = 0;
  let raf;

  function fmt(sec){
    sec = Math.max(0, Math.round(sec));
    const m = Math.floor(sec/60);
    const s = sec%60;
    return `${m}:${String(s).padStart(2,'0')}`;
  }
  function updateUI(){
    const pct = (t/total)*100;
    progressFill.style.width = pct + '%';
    elapsedEl.textContent = fmt(t);
    durationEl.textContent = '-' + fmt(total - t);
  }
  function step(){
    if(!isPlaying) return;
    t += 0.2;
    if(t > total){ t = 0; }
    updateUI();
    raf = requestAnimationFrame(step);
  }
  function play(){
    if(isPlaying) return;
    isPlaying = true;
    playPauseIcon.src = "__PLACEHOLDER_ASSETS_BASE_DIR__/008.svg"; // pause
    playPauseIcon.alt = "Pause";
    step();
  }
  function pause(){
    isPlaying = false;
    playPauseIcon.src = "__PLACEHOLDER_ASSETS_BASE_DIR__/011.svg"; // play
    playPauseIcon.alt = "Play";
    cancelAnimationFrame(raf);
  }

  playPause.addEventListener('click', ()=>{
    if(isPlaying) pause(); else play();
  });

  progressBar.addEventListener('click', (e)=>{
    const rect = progressBar.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const pct = Math.min(1, Math.max(0, x/rect.width));
    t = Math.round(total * pct);
    updateUI();
  });

  // Prev/Next just jump around the track, for demo
  document.getElementById('prevBtn').addEventListener('click', ()=>{
    t = Math.max(0, t - 10); updateUI();
  });
  document.getElementById('nextBtn').addEventListener('click', ()=>{
    t = Math.min(total, t + 10); updateUI();
  });

  // Initialize UI
  updateUI();
</script>
