<script>
/* 2048 logic (simple, single-file mock) */
(function(){
  const size = 4;
  let board = [];
  let score = 0;
  let best = Number(localStorage.getItem('best2048')||0);
  const tilesEl = document.getElementById('tiles');
  const boardEl = document.getElementById('board');
  const scoreVal = document.getElementById('scoreVal');
  const bestVal = document.getElementById('bestVal');
  const newBtn = document.getElementById('newBtn');
  bestVal.textContent = best;

  function newGame(){
    board = Array.from({length:size}, ()=>Array(size).fill(0));
    score = 0;
    updateScore(0);
    addRandomTile();
    addRandomTile();
    render();
  }

  function updateScore(add){
    if(add>0) floatScore(add);
    score += add;
    scoreVal.textContent = score;
    if(score>best){
      best = score;
      bestVal.textContent = best;
      localStorage.setItem('best2048', best);
    }
  }

  function emptyCells(){
    const arr=[];
    for(let r=0;r<size;r++)for(let c=0;c<size;c++) if(board[r][c]===0) arr.push([r,c]);
    return arr;
  }

  function addRandomTile(){
    const empties = emptyCells();
    if(empties.length===0) return false;
    const [r,c] = empties[Math.floor(Math.random()*empties.length)];
    board[r][c] = Math.random()<0.9 ? 2 : 4;
    return true;
  }

  function slideRow(row){
    const r = row.filter(n=>n!==0);
    let points = 0;
    for(let i=0;i<r.length-1;i++){
      if(r[i]===r[i+1]){
        r[i]*=2;
        points += r[i];
        r.splice(i+1,1);
      }
    }
    while(r.length<size) r.push(0);
    return {row:r, points};
  }

  function move(dir){
    let moved = false;
    let gained = 0;
    if(dir==='left'){
      for(let r=0;r<size;r++){
        const before = board[r].join(',');
        const s = slideRow(board[r]);
        board[r]=s.row;
        gained+=s.points;
        if(before!==board[r].join(',')) moved=true;
      }
    }else if(dir==='right'){
      for(let r=0;r<size;r++){
        const before = board[r].join(',');
        const rev = [...board[r]].reverse();
        const s = slideRow(rev);
        board[r]=s.row.reverse();
        gained+=s.points;
        if(before!==board[r].join(',')) moved=true;
      }
    }else if(dir==='up'){
      for(let c=0;c<size;c++){
        const col = [];
        for(let r=0;r<size;r++) col.push(board[r][c]);
        const before = col.join(',');
        const s = slideRow(col);
        for(let r=0;r<size;r++) board[r][c]=s.row[r];
        gained+=s.points;
        if(before!==s.row.join(',')) moved=true;
      }
    }else if(dir==='down'){
      for(let c=0;c<size;c++){
        const col = [];
        for(let r=0;r<size;r++) col.push(board[r][c]);
        const before = col.join(',');
        const rev = col.reverse();
        const s = slideRow(rev);
        const res = s.row.reverse();
        for(let r=0;r<size;r++) board[r][c]=res[r];
        gained+=s.points;
        if(before!==res.join(',')) moved=true;
      }
    }
    if(moved){
      updateScore(gained);
      addRandomTile();
      render(true);
      if(isGameOver()){
        setTimeout(()=>alert('Game over!'),200);
      }
    }
  }

  function isGameOver(){
    if(emptyCells().length>0) return false;
    // check neighbors
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        const v=board[r][c];
        if(r+1<size && board[r+1][c]===v) return false;
        if(c+1<size && board[r][c+1]===v) return false;
      }
    }
    return true;
  }

  function posToPixels(r,c){
    const gap = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--gap'));
    const bsize = boardEl.clientWidth;
    const cell = (bsize - gap*5)/4;
    const x = c*(cell+gap);
    const y = r*(cell+gap);
    return {x,y, size:cell};
  }

  function render(popNew=false){
    // rebuild tiles
    tilesEl.innerHTML='';
    for(let r=0;r<size;r++){
      for(let c=0;c<size;c++){
        const v = board[r][c];
        if(!v) continue;
        const div = document.createElement('div');
        const pos = posToPixels(r,c);
        div.className = `tile n${v}` + (popNew && v<=4 ? ' new' : '');
        div.style.transform = `translate(${pos.x}px, ${pos.y}px)`;
        div.textContent = v;
        tilesEl.appendChild(div);
      }
    }
  }

  // Floating score near score box
  function floatScore(n){
    if(n<=0) return;
    const wrap = document.getElementById('scoreBox');
    const s = document.createElement('div');
    s.className='pop';
    s.textContent='+'+n;
    wrap.appendChild(s);
    setTimeout(()=>s.remove(),900);
  }

  // Input handling
  function onKey(e){
    const key = e.key.toLowerCase();
    if(['arrowup','w'].includes(key)) {e.preventDefault(); move('up');}
    else if(['arrowdown','s'].includes(key)) {e.preventDefault(); move('down');}
    else if(['arrowleft','a'].includes(key)) {e.preventDefault(); move('left');}
    else if(['arrowright','d'].includes(key)) {e.preventDefault(); move('right');}
  }
  window.addEventListener('keydown', onKey);

  // Simple touch swipe
  let touchStart=null;
  boardEl.addEventListener('touchstart', e=>{
    const t = e.touches[0]; touchStart=[t.clientX, t.clientY];
  }, {passive:true});
  boardEl.addEventListener('touchend', e=>{
    if(!touchStart) return;
    const t = e.changedTouches[0];
    const dx = t.clientX - touchStart[0];
    const dy = t.clientY - touchStart[1];
    const ax = Math.abs(dx), ay = Math.abs(dy);
    if(Math.max(ax,ay) < 20) return;
    if(ax>ay) move(dx>0?'right':'left');
    else move(dy>0?'down':'up');
    touchStart=null;
  });

  newBtn.addEventListener('click', ()=>newGame());

  function mockStartingLayout(){
    // Start with a board similar to screenshots
    board = [
      [0,0,0,2],
      [0,2,4,0],
      [0,0,8,0],
      [2,8,0,2]
    ];
    score = 36;
    updateScore(0);
    render();
  }

  function mockLaunch(name){
    alert('Opening '+name+' (mock)');
  }
  window.mockLaunch = mockLaunch;

  // init
  mockStartingLayout();

  // adapt on resize (re-render to keep tile size alignment)
  window.addEventListener('resize', ()=>render());
})();
</script>
</body>
