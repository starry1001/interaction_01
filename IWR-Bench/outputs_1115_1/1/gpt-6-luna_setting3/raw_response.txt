<script>
  const boardEl = document.getElementById('board');
  const scoreEl = document.getElementById('score');
  const bestEl = document.getElementById('best');
  const plusEl = document.getElementById('plus');

  let grid = [
    [0, 0, 2, 0],
    [0, 0, 0, 0],
    [0, 2, 4, 2],
    [0, 0, 8, 2]
  ];

  let score = 20;
  let best = Number(localStorage.getItem('best2048') || 20);

  function draw() {
    boardEl.innerHTML = '';
    grid.forEach(row => row.forEach(value => {
      const cell = document.createElement('div');
      cell.className = 'cell' + (value ? ' v' + value : '');
      if (value) cell.textContent = value;
      boardEl.appendChild(cell);
    }));
    scoreEl.textContent = score;
    bestEl.textContent = best;
  }

  function emptyCells() {
    const cells = [];
    for (let r = 0; r < 4; r++) {
      for (let c = 0; c < 4; c++) {
        if (!grid[r][c]) cells.push([r, c]);
      }
    }
    return cells;
  }

  function addRandom() {
    const cells = emptyCells();
    if (!cells.length) return;
    const [r, c] = cells[Math.floor(Math.random() * cells.length)];
    grid[r][c] = Math.random() < .9 ? 2 : 4;
  }

  function slideLine(line) {
    const original = line.slice();
    let compact = line.filter(Boolean);
    let gained = 0;

    for (let i = 0; i < compact.length - 1; i++) {
      if (compact[i] === compact[i + 1]) {
        compact[i] *= 2;
        gained += compact[i];
        compact.splice(i + 1, 1);
      }
    }

    while (compact.length < 4) compact.push(0);
    return { line: compact, gained, changed: compact.some((v, i) => v !== original[i]) };
  }

  function move(direction) {
    let changed = false;
    let gained = 0;
    const next = Array.from({length: 4}, () => [0, 0, 0, 0]);

    for (let i = 0; i < 4; i++) {
      let line;
      if (direction === 'left' || direction === 'right') {
        line = grid[i].slice();
        if (direction === 'right') line.reverse();
        const result = slideLine(line);
        line = result.line;
        if (direction === 'right') line.reverse();
        for (let j = 0; j < 4; j++) next[i][j] = line[j];
        changed ||= result.changed;
        gained += result.gained;
      } else {
        line = [];
        for (let j = 0; j < 4; j++) line.push(grid[j][i]);
        if (direction === 'down') line.reverse();
        const result = slideLine(line);
        line = result.line;
        if (direction === 'down') line.reverse();
        for (let j = 0; j < 4; j++) next[j][i] = line[j];
        changed ||= result.changed;
        gained += result.gained;
      }
    }

    if (!changed) return;
    grid = next;
    score += gained;
    if (score > best) {
      best = score;
      localStorage.setItem('best2048', best);
    }
    addRandom();
    showPlus(gained);
    draw();
  }

  function showPlus(amount) {
    if (!amount) return;
    plusEl.textContent = '+' + amount;
    plusEl.classList.remove('show');
    void plusEl.offsetWidth;
    plusEl.classList.add('show');
  }

  function newGame() {
    grid = [
      [0, 0, 2, 0],
      [0, 0, 0, 0],
      [0, 2, 4, 2],
      [0, 0, 8, 2]
    ];
    score = 20;
    draw();
  }

  document.getElementById('newGame').addEventListener('click', newGame);

  document.querySelectorAll('.tab').forEach(tab => {
    tab.addEventListener('click', () => {
      document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
      tab.classList.add('active');
    });
  });

  document.addEventListener('keydown', e => {
    const keys = {
      ArrowLeft: 'left', a: 'left', A: 'left',
      ArrowRight: 'right', d: 'right', D: 'right',
      ArrowUp: 'up', w: 'up', W: 'up',
      ArrowDown: 'down', s: 'down', S: 'down'
    };
    if (keys[e.key]) {
      e.preventDefault();
      move(keys[e.key]);
    }
  });

  let touchStartX = 0;
  let touchStartY = 0;

  boardEl.addEventListener('touchstart', e => {
    const touch = e.changedTouches[0];
    touchStartX = touch.clientX;
    touchStartY = touch.clientY;
  }, {passive: true});

  boardEl.addEventListener('touchend', e => {
    const touch = e.changedTouches[0];
    const dx = touch.clientX - touchStartX;
    const dy = touch.clientY - touchStartY;
    if (Math.max(Math.abs(dx), Math.abs(dy)) < 25) return;
    if (Math.abs(dx) > Math.abs(dy)) move(dx > 0 ? 'right' : 'left');
    else move(dy > 0 ? 'down' : 'up');
  }, {passive: true});

  draw();
</script>
