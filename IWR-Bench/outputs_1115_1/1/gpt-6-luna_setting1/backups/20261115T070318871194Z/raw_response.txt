<script>
  const initialBoard = [
    [0, 0, 2, 0],
    [0, 0, 0, 0],
    [0, 2, 4, 2],
    [0, 0, 8, 2]
  ];

  let board = initialBoard.map(row => row.slice());
  let score = 20;
  let best = 20;

  const boardEl = document.getElementById('board');
  const scoreEl = document.getElementById('score');
  const bestEl = document.getElementById('best');
  const deltaEl = document.getElementById('delta');

  function render() {
    boardEl.querySelectorAll('.cell').forEach(el => el.remove());

    board.forEach(row => row.forEach(value => {
      const cell = document.createElement('div');
      cell.className = 'cell' + (value ? ' v' + value : '');
      if (value) cell.textContent = value;
      boardEl.appendChild(cell);
    }));

    scoreEl.textContent = score;
    bestEl.textContent = best;
  }

  function emptyCells() {
    const result = [];
    for (let r = 0; r < 4; r++) {
      for (let c = 0; c < 4; c++) {
        if (!board[r][c]) result.push([r, c]);
      }
    }
    return result;
  }

  function addRandomTile() {
    const spaces = emptyCells();
    if (!spaces.length) return;
    const [r, c] = spaces[Math.floor(Math.random() * spaces.length)];
    board[r][c] = Math.random() < .9 ? 2 : 4;
  }

  function slideLine(line) {
    const compact = line.filter(Boolean);
    const result = [];
    let gained = 0;

    for (let i = 0; i < compact.length; i++) {
      if (compact[i] === compact[i + 1]) {
        const merged = compact[i] * 2;
        result.push(merged);
        gained += merged;
        i++;
      } else {
        result.push(compact[i]);
      }
    }

    while (result.length < 4) result.push(0);
    return { line: result, gained };
  }

  function move(direction) {
    let changed = false;
    let gained = 0;
    const next = Array.from({ length: 4 }, () => Array(4).fill(0));

    if (direction === 'left' || direction === 'right') {
      for (let r = 0; r < 4; r++) {
        let line = board[r].slice();
        if (direction === 'right') line.reverse();
        const processed = slideLine(line);
        line = processed.line;
        if (direction === 'right') line.reverse();
        next[r] = line;
        gained += processed.gained;
      }
    } else {
      for (let c = 0; c < 4; c++) {
        let line = [board[0][c], board[1][c], board[2][c], board[3][c]];
        if (direction === 'down') line.reverse();
        const processed = slideLine(line);
        line = processed.line;
        if (direction === 'down') line.reverse();
        for (let r = 0; r < 4; r++) next[r][c] = line[r];
        gained += processed.gained;
      }
    }

    for (let r = 0; r < 4; r++) {
      for (let c = 0; c < 4; c++) {
        if (next[r][c] !== board[r][c]) changed = true;
      }
    }

    if (!changed) return;

    board = next;
    if (gained) {
      score += gained;
      best = Math.max(best, score);
      deltaEl.textContent = '+' + gained;
      deltaEl.classList.remove('show');
      void deltaEl.offsetWidth;
      deltaEl.classList.add('show');
      setTimeout(() => deltaEl.classList.remove('show'), 550);
    }
    addRandomTile();
    render();
  }

  function resetGame() {
    board = initialBoard.map(row => row.slice());
    score = 20;
    render();
  }

  document.addEventListener('keydown', e => {
    const key = e.key.toLowerCase();
    const map = {
      arrowleft: 'left', a: 'left',
      arrowright: 'right', d: 'right',
      arrowup: 'up', w: 'up',
      arrowdown: 'down', s: 'down'
    };
    if (map[key]) {
      e.preventDefault();
      move(map[key]);
    }
  });

  document.getElementById('newGame').addEventListener('click', resetGame);

  document.getElementById('classicBtn').addEventListener('click', () => {
    document.getElementById('classicBtn').classList.add('active');
    document.getElementById('cupcakesBtn').classList.remove('active');
  });

  document.getElementById('cupcakesBtn').addEventListener('click', () => {
    document.getElementById('cupcakesBtn').classList.add('active');
    document.getElementById('classicBtn').classList.remove('active');
  });

  let touchStartX = 0;
  let touchStartY = 0;

  boardEl.addEventListener('touchstart', e => {
    const touch = e.changedTouches[0];
    touchStartX = touch.clientX;
    touchStartY = touch.clientY;
  }, { passive: true });

  boardEl.addEventListener('touchend', e => {
    const touch = e.changedTouches[0];
    const dx = touch.clientX - touchStartX;
    const dy = touch.clientY - touchStartY;
    if (Math.max(Math.abs(dx), Math.abs(dy)) < 25) return;
    if (Math.abs(dx) > Math.abs(dy)) move(dx > 0 ? 'right' : 'left');
    else move(dy > 0 ? 'down' : 'up');
  }, { passive: true });

  render();
</script>
