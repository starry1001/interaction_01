const modes={
 beginner:{w:9,h:9,mines:10,cls:'small'},
 intermediate:{w:16,h:16,mines:40,cls:'medium'},
 expert:{w:30,h:16,mines:99,cls:'large'},
 custom:{w:30,h:20,mines:100,cls:'custom'}
};
let current='beginner', gameState='new', revealed={};

function makeBoard(){
 const d=modes[current];
 const wrap=document.getElementById('gameWrap');
 wrap.innerHTML='';
 const game=document.createElement('div');
 game.className='game '+d.cls;
 const panel=document.createElement('div'); panel.className='panel';
 panel.innerHTML='<div class="counter">'+String(d.mines).padStart(3,'0')+'</div><div class="face btn">🙂</div><div class="counter">000</div>';
 game.appendChild(panel);
 const grid=document.createElement('div');grid.className='grid';
 grid.style.gridTemplateColumns=`repeat(${d.w},${d.cls==='small'||d.cls==='medium'?'23px':'21px'})`;
 const opened=[[0,0,'1'],[1,0,'2'],[2,0,'2'],[3,0,'2'],[4,0,'2'],[5,0,'1']];
 for(let y=0;y<d.h;y++) for(let x=0;x<d.w;x++){
   const c=document.createElement('div');c.className='cell btn';c.dataset.x=x;c.dataset.y=y;
   const found=opened.find(a=>a[0]===x&&a[1]===y);
   if(current==='beginner' && found){c.classList.add('open',found[2]==='2'?'n2':'');c.textContent=found[2]}
   c.addEventListener('click',()=>reveal(c));
   c.addEventListener('contextmenu',e=>{e.preventDefault();c.textContent=c.textContent==='⚑'?'':'⚑'});
   grid.appendChild(c);
 }
 game.appendChild(grid);
 if(current==='beginner') {const b=document.createElement('div');b.className='bottomline';game.appendChild(b)}
 const tools=document.createElement('div');tools.className='tools';
 ['↗','♧','⚒','▣'].forEach(t=>{let q=document.createElement('button');q.className='btn tool';q.textContent=t;tools.appendChild(q)});
 wrap.appendChild(game);wrap.appendChild(tools);
 document.querySelectorAll('.tabs button').forEach(b=>b.classList.toggle('active',b.dataset.mode===current));
}
function reveal(c){
 if(c.classList.contains('open'))return;
 c.classList.add('open');
 if(Math.random()<.08){c.textContent='✹';c.classList.add('mine')}
 else {let n=Math.ceil(Math.random()*3);c.textContent=n;c.classList.add(n===2?'n2':n===3?'n3':'')}
}
document.querySelectorAll('.tabs button').forEach(b=>b.addEventListener('click',()=>{
 current=b.dataset.mode;
 document.getElementById('customControls').classList.toggle('hidden',current!=='custom');
 document.getElementById('stats').classList.add('hidden');
 if(current==='custom'){
   modes.custom.w=parseInt(width.value)||30;modes.custom.h=parseInt(height.value)||20;modes.custom.mines=parseInt(mines.value)||100;
 }
 makeBoard();
}));
document.getElementById('update').addEventListener('click',()=>{
 modes.custom.w=Math.max(5,parseInt(width.value)||30);
 modes.custom.h=Math.max(5,parseInt(height.value)||20);
 modes.custom.mines=Math.max(1,parseInt(mines.value)||100);
 makeBoard();
});
document.addEventListener('click',e=>{
 if(e.target.classList.contains('face')){makeBoard()}
});
makeBoard();
</script>
