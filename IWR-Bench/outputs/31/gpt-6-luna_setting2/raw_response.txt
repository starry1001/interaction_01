  const input = document.getElementById('commentBox');
  const send = document.getElementById('sendComment');

    const value = input.value.trim();
    if (!value) return;
    const activity = document.querySelector('.activity-scroll');
    const card = document.createElement('article');
    card.className = 'comment-card';
    card.innerHTML = '<div class="comment-meta"><span class="avatar">朱</span><b>朱嘉亮</b><span class="comment-time">Just now</span></div><div class="comment-text"></div><div class="comment-footer"><span>♧ &nbsp;☻</span><button class="btn" style="border:0;background:none;color:#777">Reply</button></div>';
    card.querySelector('.comment-text').textContent = value;
    activity.appendChild(card);
    input.value = '';
  });
  input.addEventListener('keydown', e => {
    if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); send.click(); }
  });
  document.querySelectorAll('.status-pill').forEach(el => el.addEventListener('click', () => {
    el.textContent = el.textContent === 'COMPLETE' ? 'IN PROGRESS' : 'COMPLETE';
