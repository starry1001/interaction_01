<script>
(function(){
  const list=document.getElementById('exprList'), current=document.getElementById('current');
  const currentAnswer=document.getElementById('currentAnswer'), error=document.getElementById('error');
  const calc=document.querySelector('.calculator'), graph=document.querySelector('.graph');
  let expr='', rows=[], lastAns=0, special=null;

  function pretty(s, editing=false){
    if(!s) return editing?'<span class="caret"></span>':'';
    let out=s;
    out=out.replace(/sqrt\(([^()]*)\)/g,'<span style="font-size:22px">√</span><span style="border-top:1px solid #444;padding-top:1px">$1</span>');
    out=out.replace(/abs\(([^()]*)\)/g,'|$1|');
    out=out.replace(/(\d+)\^(\d+)/g,'$1<sup>$2</sup>');
    out=out.replace(/(\d+)\^/g,'$1<sup><span class="caret"></span></sup>');
    out=out.replace(/\(([^()/]+)\/([^()/]*)\)/g,'<span class="fraction"><span class="num">$1</span><span class="den">$2</span></span>');
    out=out.replace(/(\d+)\/(\d+)/g,'<span class="fraction"><span class="num">$1</span><span class="den">$2</span></span>');
    out=out.replace(/\*/g,'×').replace(/\//g,'÷').replace(/-/g,'−');
    return out+(editing?'<span class="caret"></span>':'');
  }
  function value(s){
    if(!s)return '';
    try{
      let t=s.replace(/ans/g,String(lastAns)).replace(/sqrt/g,'Math.sqrt').replace(/abs/g,'Math.abs');
      t=t.replace(/\^/g,'**');
      let v=Function('return '+t)();
      if(typeof v==='number' && isFinite(v)) return Math.round(v*1e10)/1e10;
    }catch(e){}
    return '';
  }
  function render(){
    list.innerHTML=rows.map(r=>`<div class="row"><div class="math">${pretty(r.e)}</div><div class="answer">= ${r.a}</div></div>`).join('');
    current.innerHTML=pretty(expr,true);
    let a=value(expr); currentAnswer.textContent=(expr&&a!==''?'= '+a:'');
    error.textContent=(expr && !a && /[+\-*/^]$/.test(expr)?'':'');
    if(expr && expr==='-' ) error.textContent='⚠';
    document.querySelector('.hint').style.display=rows.length?'block':'none';
  }
  function add(v){expr+=v;render()}
  function action(a){
    if(a==='square'){ if(expr) expr+='^2'; else expr='^'; }
    else if(a==='power'){expr+='^'}
    else if(a==='sqrt'){expr+='sqrt('}
    else if(a==='abs'){expr+='abs('}
    else if(a==='frac'){expr+='('}
    else if(a==='left'){expr+='('}
    else if(a==='right'){expr+=')'}
    else if(a==='back'){expr=expr.slice(0,-1)}
    else if(a==='enter'){
      if(expr){
        let v=value(expr);
        if(v!==''){rows.push({e:expr,a:v});lastAns=v;expr=''}
      }
    }
    render();
  }
  document.querySelectorAll('[data-value]').forEach(b=>b.addEventListener('click',()=>add(b.dataset.value)));
  document.querySelectorAll('[data-action]').forEach(b=>b.addEventListener('click',()=>action(b.dataset.action)));
  document.getElementById('clearBtn').addEventListener('click',()=>{rows=[];expr='';lastAns=0;render()});
  document.addEventListener('keydown',e=>{
    if(/[0-9.+\-*/()]/.test(e.key))add(e.key);
    else if(e.key==='Enter')action('enter');
    else if(e.key==='Backspace')action('back');
  });
  render();
})();
</script>
