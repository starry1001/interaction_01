    document.getElementById('modalshade').classList.add('open');
  }
  function closeModal(add){
    if(add){
      if(checked.has('one sided headache')&&!symptoms.includes('one sided headache'))symptoms.push('one sided headache');
      if(checked.has('headache')&&!symptoms.includes('headache'))symptoms.push('headache');
      if(!symptoms.length)symptoms.push(...Array.from(checked).filter(x=>x!=='None of the above'));
      query='';
    }
    document.getElementById('modalshade').classList.remove('open');render();
  }
  document.getElementById('done').onclick=()=>closeModal(true);
  document.getElementById('skip').onclick=()=>closeModal(false);
