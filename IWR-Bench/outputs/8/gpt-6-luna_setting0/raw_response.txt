<script>
  const countInput = document.getElementById("count");
  const diceArea = document.getElementById("diceArea");
  const totalEl = document.getElementById("total");
  const plus = document.getElementById("plus");
  const minus = document.getElementById("minus");
  const roll = document.getElementById("roll");
  const counter = document.getElementById("counter");

  let count = 1;
  let rollIndex = 0;

  const sequences = [
    [1, 5, 5, 1],
    [2, 1, 8, 3],
    [1, 7, 8, 6]
  ];

  function makeDie(value) {
    const die = document.createElement("div");
    die.className = "die";
    const span = document.createElement("span");
    span.textContent = value;
    die.appendChild(span);
    return die;
  }

  function render(values) {
    diceArea.innerHTML = "";
    diceArea.className = "dice-area" + (values.length === 1 ? " single-die" : "");
    values.forEach(v => diceArea.appendChild(makeDie(v)));
    totalEl.textContent = values.reduce((a, b) => a + b, 0);
  }

  function updateCount() {
    countInput.value = count;
    counter.style.outline = "2px dashed #49369a";
    counter.style.outlineOffset = "3px";
    setTimeout(() => {
      counter.style.outline = "";
      counter.style.outlineOffset = "";
    }, 180);
  }

  plus.addEventListener("click", () => {
    if (count < 20) count++;
    updateCount();
  });

  minus.addEventListener("click", () => {
    if (count > 1) count--;
    updateCount();
  });

  roll.addEventListener("click", () => {
    let values;
    if (count === 1) {
      values = [7];
    } else if (count === 4) {
      values = sequences[rollIndex % sequences.length];
      rollIndex++;
    } else {
      values = Array.from({length: count}, () => Math.floor(Math.random() * 8) + 1);
    }
    render(values);
    roll.style.outline = "2px dashed #49369a";
    roll.style.outlineOffset = "3px";
    setTimeout(() => {
      roll.style.outline = "";
      roll.style.outlineOffset = "";
    }, 220);
  });

  document.getElementById("modes").addEventListener("click", function() {
    document.body.classList.toggle("dark-ish");
  });

  render([7]);
</script>
</body>
