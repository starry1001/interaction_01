<script>
const editor=document.getElementById('editor');
const preview=document.getElementById('preview');
const cursorInfo=document.getElementById('cursor');

function esc(s){
  return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
}
function inline(s){
  s=esc(s);
  s=s.replace(/`([^`]+)`/g,'<code>$1</code>');
  s=s.replace(/\*\*([^*]+)\*\*/g,'<strong>$1</strong>');
  s=s.replace(/__([^_]+)__/g,'<strong>$1</strong>');
  s=s.replace(/~~([^~]+)~~/g,'<del>$1</del>');
  s=s.replace(/\*([^*]+)\*/g,'<em>$1</em>');
  s=s.replace(/_([^_]+)_/g,'<em>$1</em>');
  return s;
}
function render(md){
  const lines=md.split('\n'), out=[]; let para=[], list=null;
  function flush(){
    if(para.length){out.push('<p>'+inline(para.join('<br>'))+'</p>');para=[]}
    if(list){out.push(list.type+'>'+list.items.join('')+'</'+(list.type==='ul'?'ul':'ol')+'>');list=null}
  }
  for(let line of lines){
    let m;
    if((m=line.match(/^###\s+(.*)$/))){flush();out.push('<h3>'+inline(m[1])+'</h3>');continue}
    if((m=line.match(/^##\s+(.*)$/))){flush();out.push('<h2>'+inline(m[1])+'</h2>');continue}
    if((m=line.match(/^#\s+(.*)$/))){flush();out.push('<h1>'+inline(m[1])+'</h1>');continue}
    if(/^(\*\*\*|---|___)\s*$/.test(line)){flush();out.push('<hr>');continue}
    if((m=line.match(/^\s*[-*]\s+(.*)$/))){if(!list||list.type!=='ul'){flush();list={type:'ul',items:[]}}list.items.push('<li>'+inline(m[1])+'</li>');continue}
    if((m=line.match(/^\s*\d+\.\s+(.*)$/))){if(!list||list.type!=='ol'){flush();list={type:'ol',items:[]}}list.items.push('<li>'+inline(m[1])+'</li>');continue}
    if(line.trim()===''){flush();continue}
    para.push(line)
  }
  flush(); preview.innerHTML=out.join('');
}
function update(){
  render(editor.value);
  const text=editor.value;
  document.getElementById('bytes').textContent=text.length+' bytes';
  document.getElementById('chars').textContent=text.length+' characters';
  document.getElementById('words').textContent=(text.trim().match(/\S+/g)||[]).length+' words';
  document.getElementById('lines').textContent=text.split('\n').length+' lines';
  const before=text.slice(0,editor.selectionStart), last=before.lastIndexOf('\n');
  cursorInfo.textContent='Ln '+(before.split('\n').length)+', Col '+(before.length-last);
}
editor.addEventListener('input',update);
editor.addEventListener('keyup',update);
editor.addEventListener('click',update);
editor.addEventListener('select',update);

function selectedRange(){
  let a=editor.selectionStart,b=editor.selectionEnd;
  if(a===b){
    const lineStart=editor.value.lastIndexOf('\n',a-1)+1;
    const lineEnd=editor.value.indexOf('\n',a);
    a=lineStart;b=lineEnd<0?editor.value.length:lineEnd;
  }
  return [a,b,editor.value.slice(a,b)];
}
function replaceSelection(before,after=before){
  const [a,b,t]=selectedRange();
  const val=editor.value;
  editor.value=val.slice(0,a)+before+t+after+val.slice(b);
  editor.focus();
  editor.setSelectionRange(a+before.length,a+before.length+t.length);
  update();
}
function prefixLines(prefix){
  const [a,b,t]=selectedRange();
  const start=editor.value.lastIndexOf('\n',a-1)+1;
  const chunk=editor.value.slice(start,b);
  const changed=chunk.split('\n').map(x=>prefix+x).join('\n');
  editor.value=editor.value.slice(0,start)+changed+editor.value.slice(b);
  editor.focus();editor.setSelectionRange(start,start+changed.length);update();
}
document.querySelectorAll('[data-action]').forEach(btn=>{
  btn.addEventListener('click',()=>{
    const act=btn.dataset.action;
    if(act==='bold')replaceSelection('**');
    else if(act==='italic')replaceSelection('*');
    else if(act==='strike')replaceSelection('~~');
    else if(act==='code')replaceSelection('`');
    else if(act==='heading')prefixLines('## ');
    else if(act==='quote')prefixLines('> ');
    else if(act==='ul')prefixLines('- ');
    else if(act==='ol')prefixLines('1. ');
    else if(act==='check')prefixLines('- [ ] ');
    else if(act==='link')replaceSelection('[','](https://example.com)');
    else if(act==='image')replaceSelection('![','](image.png)');
    else if(act==='table'){
      const [a,b,t]=selectedRange();
      replaceSelection('| '+t+' |',' |\\n|---|');
    }
  });
});
update();
