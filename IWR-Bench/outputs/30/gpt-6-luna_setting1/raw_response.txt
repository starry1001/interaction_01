const source=document.getElementById('source');
const preview=document.getElementById('preview');
const toast=document.getElementById('toast');

function esc(s){return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')}
function inline(s){
  return esc(s)
    .replace(/\*\*(.+?)\*\*/g,'<strong>$1</strong>')
    .replace(/__(.+?)__/g,'<strong>$1</strong>')
    .replace(/\*(.+?)\*/g,'<em>$1</em>')
    .replace(/_(.+?)_/g,'<em>$1</em>')
    .replace(/~~(.+?)~~/g,'<del>$1</del>');
}
function render(md){
  const lines=md.replace(/\r/g,'').split('\n'), out=[];
  let i=0;
  while(i<lines.length){
    let l=lines[i];
    if(!l.trim()){i++;continue}
    if(/^### /.test(l)){out.push('<h3>'+inline(l.slice(4))+'</h3>');i++;continue}
    if(/^## /.test(l)){out.push('<h2>'+inline(l.slice(3))+'</h2>');i++;continue}
    if(/^# /.test(l)){out.push('<h1>'+inline(l.slice(2))+'</h1>');i++;continue}
    if(/^---+$/.test(l)){out.push('<hr>');i++;continue}
    if(/^[-*] /.test(l)){
      let a='<ul>'; while(i<lines.length&&/^[-*] /.test(lines[i])){a+='<li>'+inline(lines[i].slice(2))+'</li>';i++}
      out.push(a+'</ul>');continue
    }
    let p=inline(l);i++;
    while(i<lines.length&&lines[i].trim()&&!/^(#{1,3} |[-*] |---+$)/.test(lines[i])){
      p+='<br>'+inline(lines[i]);i++;
    }
    out.push('<p>'+p+'</p>');
  }
  preview.innerHTML=out.join('');
}
function update(){
  render(source.value);
  const pos=source.selectionStart;
  const before=source.value.slice(0,pos);
  const line=before.split('\n').length;
  const col=before.length-before.lastIndexOf('\n')-1;
  const words=source.value.trim().split(/\s+/).filter(Boolean).length;
  const lines=source.value.split('\n').length;
  document.getElementById('statusLeft').innerHTML='Markdown&nbsp;&nbsp;'+source.value.length+' bytes&nbsp;&nbsp;'+words+' words&nbsp;&nbsp;'+lines+' lines&nbsp;&nbsp;Ln '+line+', Col '+col;
}
function wrap(a,b){
  const s=source.selectionStart,e=source.selectionEnd,v=source.value;
  const selected=v.slice(s,e);
  source.setRangeText(a+selected+b,s,e,'select');
  update();source.focus();
}
function linePrefix(prefix){
  const s=source.selectionStart,e=source.selectionEnd,v=source.value;
  const start=v.lastIndexOf('\n',s-1)+1;
  const end=v.indexOf('\n',e)<0?v.length:v.indexOf('\n',e);
  const text=v.slice(start,end);
  source.setRangeText(prefix+text,start,end,'select');update();source.focus();
}
document.querySelectorAll('[data-action]').forEach(b=>{
  b.addEventListener('mousedown',e=>e.preventDefault());
  b.addEventListener('click',()=>{
    const a=b.dataset.action;
    if(a==='bold')wrap('**','**');
    if(a==='italic')wrap('*','*');
    if(a==='strike')wrap('~~','~~');
    if(a==='heading')linePrefix('## ');
    if(a==='ul')linePrefix('- ');
    if(a==='ol')linePrefix('1. ');
    if(a==='check')linePrefix('- [ ] ');
    if(['bold','italic','strike'].includes(a)){b.classList.add('active');setTimeout(()=>b.classList.remove('active'),250);toast.classList.add('show');setTimeout(()=>toast.classList.remove('show'),2200)}
  });
});
source.addEventListener('input',update);
source.addEventListener('click',update);
source.addEventListener('keyup',update);
document.querySelector('.folder').addEventListener('click',()=>toast.classList.toggle('show'));
document.querySelectorAll('.split-tools button').forEach((b,i)=>{
  b.addEventListener('click',()=>{
    const ed=document.querySelector('.editor'),pv=document.querySelector('.preview');
    if(i===0){ed.style.width='100%';pv.style.display='none'}
    if(i===1){ed.style.width='50%';pv.style.display='block'}
    if(i===2){ed.style.display='none';pv.style.width='100%'}
  })
});
render(source.value);update();
</script>
