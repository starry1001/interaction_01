<script>
(() => {
  const boardEl = document.getElementById('board');
  const capturesEl = document.getElementById('captures');
  const files = ['h','g','f','e','d','c','b','a'];
  const glyphs = {
    wk: '♔', wq: '♕', wr: '♖', wb: '♗', wn: '♘', wp: '♙',
    bk: '♚', bq: '♛', br: '♜', bb: '♝', bn: '♞', bp: '♟'
  };
  const board = Array.from({length:8}, () => Array(8).fill(null));
  const setup = [
    ['wr','wn','wb','wk','wq','wb','wn','wr'],
    Array(8).fill('wp'),
    Array(8).fill(null),
    Array(8).fill(null),
    Array(8).fill(null),
    Array(8).fill(null),
    Array(8).fill('bp'),
    ['br','bn','bb','bk','bq','bb','bn','br']
  ];
  setup.forEach((row,r) => row.forEach((p,c) => board[r][c] = p));
  // Recreate the opening move shown in the game: the white knight develops to f3.
  board[0][1] = null;
  board[2][2] = 'wn';

  let lastMove = {from:[0,1], to:[2,2]};
  let selected = null;
  let captured = [];
  let moveCount = 1;

  function render() {
    boardEl.innerHTML = '';
    for (let r=0; r<8; r++) {
      for (let c=0; c<8; c++) {
        const sq = document.createElement('div');
        sq.className = 'square btn ' + ((r+c)%2 === 0 ? 'light' : 'dark');
        sq.dataset.r = r;
        sq.dataset.c = c;

        if (lastMove && lastMove.to[0]===r && lastMove.to[1]===c) sq.classList.add('last');
        if (lastMove && lastMove.from[0]===r && lastMove.from[1]===c) sq.classList.add('last-source');
        if (selected && selected[0]===r && selected[1]===c) sq.classList.add('selected');

        if (c===0) {
          const rank = document.createElement('span');
          rank.className = 'rank-label';
          rank.textContent = String(r+1);
          sq.appendChild(rank);
        }
        if (r===7) {
          const file = document.createElement('span');
          file.className = 'file-label';
          file.textContent = files[c];
          sq.appendChild(file);
        }

        const piece = board[r][c];
        if (piece) {
          const span = document.createElement('span');
          span.className = 'piece ' + (piece[0] === 'w' ? 'white' : 'black');
          span.textContent = glyphs[piece];
          sq.appendChild(span);
        }
        sq.addEventListener('click', onSquareClick);
        boardEl.appendChild(sq);
      }
    }
    renderCaptures();
  }

  function renderCaptures() {
    capturesEl.innerHTML = '';
    captured.forEach(p => {
      const s = document.createElement('span');
      s.className = 'captured-piece ' + (p[0] === 'w' ? 'white' : 'black');
      s.textContent = glyphs[p];
      capturesEl.appendChild(s);
    });
    if (captured.length) {
      const count = document.createElement('span');
      count.textContent = '+' + captured.length;
      capturesEl.appendChild(count);
    }
  }

  function onSquareClick(e) {
    const r = Number(e.currentTarget.dataset.r);
    const c = Number(e.currentTarget.dataset.c);
    const here = board[r][c];

    if (selected) {
      const [sr,sc] = selected;
      if (sr === r && sc === c) {
        selected = null;
        render();
        return;
      }
      if (here && board[sr][sc] && here[0] === board[sr][sc][0]) {
        selected = [r,c];
        render();
        return;
      }
      const moving = board[sr][sc];
      if (moving) {
        if (here) captured.push(here);
        board[r][c] = moving;
        board[sr][sc] = null;
        if (moving === 'wp' && r === 7) board[r][c] = 'wq';
        if (moving === 'bp' && r === 0) board[r][c] = 'bq';
        lastMove = {from:[sr,sc],to:[r,c]};
        selected = null;
        moveCount++;
        updateClocks();
        render();
        return;
      }
    }
    if (here) selected = [r,c];
    else selected = null;
    render();
  }

  function updateClocks() {
    const left = document.getElementById('clock-left');
    const right = document.getElementById('clock-right');
    left.classList.toggle('active', moveCount % 2 === 1);
    right.classList.toggle('active', moveCount % 2 === 0);
    const l = (6.9 + Math.floor(moveCount / 2) * 2.1).toFixed(1);
    const rr = (1 + Math.floor((moveCount - 1) / 2) * 2).toFixed(1);
    left.textContent = l + 's';
    right.textContent = rr + 's';
  }

  render();
})();
</script>
