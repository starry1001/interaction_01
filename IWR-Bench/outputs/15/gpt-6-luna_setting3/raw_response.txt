<script>
  const first = document.getElementById('first');
  const last = document.getElementById('last');
  const email = document.getElementById('email');
  const firstField = document.getElementById('firstField');
  const lastField = document.getElementById('lastField');
  const emailField = document.getElementById('emailField');
  const emailHint = document.getElementById('emailHint');
  const emailError = document.getElementById('emailError');

  function markName(input, box) {
    if (input.value.trim()) box.classList.add('valid');
    else box.classList.remove('valid');
  }
  first.addEventListener('input', () => markName(first, firstField));
  last.addEventListener('input', () => markName(last, lastField));

  email.addEventListener('input', () => {
    emailField.classList.remove('invalid');
    emailError.style.display = 'none';
    if (email.value.trim()) {
      emailHint.style.display = 'block';
      emailHint.innerHTML = "We'll send your confirmation details to <b>" + email.value + "</b>";
    } else {
      emailHint.style.display = 'none';
    }
  });
  email.addEventListener('blur', () => {
    if (email.value.trim() && !/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.value.trim())) {
      emailField.classList.add('invalid');
      emailError.style.display = 'block';
      emailHint.style.display = 'none';
    }
  });
  email.addEventListener('focus', () => {
    if (emailField.classList.contains('invalid')) {
      emailField.classList.remove('invalid');
      emailError.style.display = 'none';
    }
  });
</script>
