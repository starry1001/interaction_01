<script>
  const first=document.getElementById('first'), last=document.getElementById('last');
  const email=document.getElementById('email'), phone=document.getElementById('phone');
  const firstField=document.getElementById('firstField'), lastField=document.getElementById('lastField');
  const emailField=document.getElementById('emailField'), hint=document.getElementById('emailHint');
  const error=document.getElementById('emailError'), emailValue=document.getElementById('emailValue');

  function markName(input, box){
    if(input.value.trim()) box.classList.add('valid'); else box.classList.remove('valid');
  }
  first.addEventListener('input',()=>markName(first,firstField));
  last.addEventListener('input',()=>markName(last,lastField));

  email.addEventListener('focus',()=>{
    if(email.value.trim() && !email.value.includes('@')){
      hint.style.display='block'; error.style.display='none'; emailField.classList.remove('email-error');
      emailValue.textContent=email.value;
    }
  });
  email.addEventListener('input',()=>{
    const v=email.value.trim();
    if(v && !v.includes('@')){
      hint.style.display='block'; error.style.display='none'; emailField.classList.remove('email-error'); emailValue.textContent=v;
    } else {
      hint.style.display='none';
    }
  });
  email.addEventListener('blur',()=>{
    const v=email.value.trim();
    if(v && !v.includes('@')){
      hint.style.display='none'; error.style.display='block'; emailField.classList.add('email-error');
    }
  });
  phone.addEventListener('focus',()=>{
    const v=email.value.trim();
    if(v && !v.includes('@')){
      hint.style.display='none'; error.style.display='block'; emailField.classList.add('email-error');
    }
  });
</script>
