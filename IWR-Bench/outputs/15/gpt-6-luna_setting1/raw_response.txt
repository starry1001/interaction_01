<script>
  const first=document.getElementById('first');
  const last=document.getElementById('last');
  const email=document.getElementById('email');
  const phone=document.getElementById('phone');
  const firstField=document.getElementById('firstField');
  const lastField=document.getElementById('lastField');
  const emailField=document.getElementById('emailField');
  const helper=document.getElementById('emailHelper');

  function validMark(input, box){
    if(input.value.trim()) box.classList.add('valid');
    else box.classList.remove('valid');
  }
  first.addEventListener('blur',()=>validMark(first,firstField));
  last.addEventListener('blur',()=>validMark(last,lastField));
  first.addEventListener('input',()=>{if(firstField.classList.contains('valid')) validMark(first,firstField)});
  last.addEventListener('input',()=>{if(lastField.classList.contains('valid')) validMark(last,lastField)});

  email.addEventListener('input',()=>{
    emailField.classList.remove('email-error');
    if(email.value.trim()){
      helper.style.display='block';
      helper.className='helper';
      helper.textContent="We'll send your confirmation details to " + email.value.trim();
    }else helper.style.display='none';
  });
  email.addEventListener('blur',()=>{
    if(email.value.trim() && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.value.trim())){
      emailField.classList.add('email-error');
      helper.style.display='block';
      helper.className='helper error';
      helper.textContent='Enter a valid email address';
    }
  });
</script>
