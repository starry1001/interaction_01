(function(){
  const display=document.getElementById('display');
  let current='0', stored=null, operator=null, waiting=false, memory=0;
  function show(){display.textContent=current}
  function format(n){return String(n).replace(/\.0+$|(\.\d*?)0+$/,'$1')}
  function calculate(a,b,op){
    a=Number(a);b=Number(b);
    if(op==='add')return a+b;
    if(op==='subtract')return a-b;
    if(op==='multiply')return a*b;
    if(op==='divide')return b===0?0:a/b;
    return b;
  }
  function value(v){
    if(waiting){current=v;waiting=false}else current=current==='0'?v:current+v;
    show();
  }
  function action(a){
    if(['add','subtract','multiply','divide'].includes(a)){
      stored=current;operator=a;waiting=true;return;
    }
    if(a==='equals' && operator){
      current=format(calculate(stored,current,operator));operator=null;stored=null;waiting=true;show();return;
    }
    if(a==='clear'){current='0';stored=null;operator=null;waiting=false;show();return}
    if(a==='decimal'){if(waiting){current='0';waiting=false}if(!current.includes('.'))current+='.';show();return}
    if(a==='sign'){current=String(Number(current)*-1);show();return}
    if(a==='sqrt'){current=format(Math.sqrt(Number(current)));show();return}
    if(a==='inverse'){current=format(1/Number(current));show();return}
    if(a==='percent'){current=format(Number(current)/100);show();return}
    if(a==='mc'){memory=0;return}
    if(a==='mr'){current=format(memory);show();return}
    if(a==='mplus'){memory+=Number(current);return}
    if(a==='mminus'){memory-=Number(current);return}
  }
  document.querySelectorAll('.key').forEach(btn=>{
    btn.addEventListener('click',()=>{
      btn.classList.add('active');setTimeout(()=>btn.classList.remove('active'),120);
      if(btn.dataset.value!==undefined)value(btn.dataset.value);else action(btn.dataset.action);
    });
  });
})();
