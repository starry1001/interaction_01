(function(){
  const display=document.getElementById('display');
  let current='0', stored=null, operator=null, fresh=true, memory=0;
  function show(v){display.textContent=String(v).slice(0,14)}
  function calculate(a,b,op){
    a=Number(a);b=Number(b);
    if(op==='+')return a+b;
    if(op==='-')return a-b;
    if(op==='*')return a*b;
    if(op==='/')return b===0?'Error':a/b;
    return b;
  }
  document.querySelectorAll('.key').forEach(btn=>{
    btn.addEventListener('click',()=>{
      const val=btn.dataset.value, action=btn.dataset.action;
      if(val!==undefined){
        if(fresh||current==='0'&&val!=='.'){current=val==='.'?'0.':val;fresh=false}
        else if(val==='.'&&!current.includes('.'))current+='.';
        else if(val!=='.')current+=val;
        show(current);return;
      }
      if(action==='clear'){current='0';stored=null;operator=null;fresh=true;show(current)}
      else if(action==='sign'){current=String(Number(current)*-1);show(current)}
      else if(action==='sqrt'){current=String(Math.sqrt(Number(current)));fresh=true;show(current)}
      else if(action==='inverse'){current=String(1/Number(current));fresh=true;show(current)}
      else if(action==='percent'){current=String(Number(current)/100);show(current)}
      else if(['+','-','*','/'].includes(val)){}
      else if(action==='equals'){
        if(operator!==null){current=String(calculate(stored,current,operator));stored=null;operator=null;fresh=true;show(current)}
      } else if(action==='mc'){memory=0}
      else if(action==='mr'){current=String(memory);fresh=true;show(current)}
      else if(action==='mplus'){memory+=Number(current)}
      else if(action==='mminus'){memory-=Number(current)}
    });
  });
  document.querySelectorAll('[data-value]').forEach(btn=>{
    if(['+','-','*','/'].includes(btn.dataset.value))btn.addEventListener('click',()=>{
      stored=Number(current);operator=btn.dataset.value;fresh=true;
    });
  });
})();
