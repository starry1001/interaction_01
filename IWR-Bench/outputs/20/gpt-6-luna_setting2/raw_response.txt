  const video = document.getElementById('shortVideo');

  video.addEventListener('loadedmetadata', () => {
    try { video.currentTime = Math.min(7.5, Math.max(0, video.duration * .55)); } catch(e) {}
  });

    if (video.duration) video.currentTime = Math.min(video.duration - .2, video.currentTime + 3);
  });

    if (video.duration) video.currentTime = Math.max(0, video.currentTime - 3);
