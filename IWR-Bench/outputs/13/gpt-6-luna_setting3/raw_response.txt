const overlay=document.getElementById('overlay');
const modal=document.querySelector('.modal');
const tabs=[...document.querySelectorAll('.tab')];
function openModal(target){
  overlay.classList.add('open');
  document.body.style.overflow='hidden';
  setTimeout(()=>scrollToSection(target),30);
}
function scrollToSection(target){
  const el=document.getElementById(target);
  if(!el)return;
  tabs.forEach(t=>t.classList.toggle('active',t.dataset.target===target));
  modal.scrollTo({top:el.offsetTop-72,behavior:'smooth'});
}
document.querySelectorAll('.hl[data-tab]').forEach(x=>x.addEventListener('click',()=>openModal(x.dataset.tab)));
tabs.forEach(t=>t.addEventListener('click',()=>scrollToSection(t.dataset.target)));

  overlay.classList.remove('open');document.body.style.overflow='';
