<script>
(function(){
  const boardEl=document.getElementById('board');
  const scoreEl=document.getElementById('score');
  const bestEl=document.getElementById('best');
  const newBtn=document.getElementById('newGame');
  const classicBtn=document.getElementById('classicBtn');
  const cupcakeBtn=document.getElementById('cupcakeBtn');

  let board, score=20;
  let best=Number(localStorage.getItem('best2048')||20);

  const initial=[
    [0,0,2,0],
    [0,0,0,0],
    [0,2,4,2],
    [0,0,8,2]
  ];

  function clone(a){return a.map(r=>r.slice())}
  function reset(){
    board=clone(initial);
    score=20;
    render();
  }
  function render(){
    boardEl.innerHTML='';
    board.flat().forEach(v=>{
      const cell=document.createElement('div');
      cell.className='cell';
      if(v){
        cell.classList.add('tile-'+v);
        cell.textContent=v;
      }
      boardEl.appendChild(cell);
    });
    scoreEl.textContent=score;
    bestEl.textContent=Math.max(best,score);
  }
  function addRandom(){
    const empty=[];
    for(let r=0;r<4;r++)for(let c=0;c<4;c++)if(!board[r][c])empty.push([r,c]);
    if(!empty.length)return;
    const p=empty[Math.floor(Math.random()*empty.length)];
    board[p[0]][p[1]]=Math.random()<.9?2:4;
  }
  function mergeLine(line){
    const vals=line.filter(Boolean), out=[], gained=0;
    for(let i=0;i<vals.length;i++){
      if(vals[i]===vals[i+1]){
        out.push(vals[i]*2);
        gained+=vals[i]*2;
        i++;
      }else out.push(vals[i]);
    }
    while(out.length<4)out.push(0);
    return {line:out,gained};
  }
  function move(dir){
    let changed=false, gained=0, next=Array.from({length:4},()=>Array(4).fill(0));
    if(dir==='left'||dir==='right'){
      for(let r=0;r<4;r++){
        let line=board[r].slice();
        if(dir==='right')line.reverse();
        const m=mergeLine(line);
        if(dir==='right')m.line.reverse();
        next[r]=m.line;
        gained+=m.gained;
      }
    }else{
      for(let c=0;c<4;c++){
        let line=[];
        for(let r=0;r<4;r++)line.push(board[r][c]);
        if(dir==='down')line.reverse();
        const m=mergeLine(line);
        if(dir==='down')m.line.reverse();
        for(let r=0;r<4;r++)next[r][c]=m.line[r];
        gained+=m.gained;
      }
    }
    changed=JSON.stringify(next)!==JSON.stringify(board);
    if(!changed)return;
    board=next;
    score+=gained;
    if(score>best){
      best=score;
      localStorage.setItem('best2048',best);
    }
    addRandom();
    render();
  }
  function key(e){
    const k=e.key.toLowerCase();
    const map={arrowleft:'left',a:'left',arrowright:'right',d:'right',arrowup:'up',w:'up',arrowdown:'down',s:'down'};
    if(map[k]){
      e.preventDefault();
      move(map[k]);
    }
  }
  document.addEventListener('keydown',key);
  newBtn.addEventListener('click',reset);
  classicBtn.addEventListener('click',()=>{
    classicBtn.classList.add('active'); cupcakeBtn.classList.remove('active');
  });
  cupcakeBtn.addEventListener('click',()=>{
    cupcakeBtn.classList.add('active'); classicBtn.classList.remove('active');
  });

  let sx=0,sy=0;
  boardEl.addEventListener('touchstart',e=>{
    const t=e.changedTouches[0]; sx=t.clientX; sy=t.clientY;
  },{passive:true});
  boardEl.addEventListener('touchend',e=>{
    const t=e.changedTouches[0], dx=t.clientX-sx, dy=t.clientY-sy;
    if(Math.max(Math.abs(dx),Math.abs(dy))<25)return;
    if(Math.abs(dx)>Math.abs(dy))move(dx>0?'right':'left');
    else move(dy>0?'down':'up');
  },{passive:true});

  reset();
})();
