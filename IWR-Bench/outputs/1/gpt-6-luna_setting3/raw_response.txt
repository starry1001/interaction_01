<script>
(function(){
  const boardEl = document.getElementById('board');
  const scoreEl = document.getElementById('score');
  const bestEl = document.getElementById('best');

  let grid = [
    [0,0,2,0],
    [0,0,0,0],
    [0,2,4,2],
    [0,0,8,2]
  ];
  let score = 20;
  let best = Number(localStorage.getItem('2048-best') || 20);
  bestEl.textContent = best;

  function render(){
    boardEl.innerHTML = '';
    grid.forEach(row => row.forEach(value => {
      const tile = document.createElement('div');
      tile.className = 'tile' + (value ? ' v' + value : '');
      if(value) tile.textContent = value;
      boardEl.appendChild(tile);
    }));
    scoreEl.textContent = score;
    bestEl.textContent = best;
  }

  function emptyCells(){
    const cells=[];
    for(let r=0;r<4;r++) for(let c=0;c<4;c++)
      if(!grid[r][c]) cells.push([r,c]);
    return cells;
  }

  function addRandom(){
    const cells=emptyCells();
    if(!cells.length)return;
    const [r,c]=cells[Math.floor(Math.random()*cells.length)];
    grid[r][c]=Math.random()<.9?2:4;
  }

  function transpose(a){
    return a[0].map((_,c)=>a.map(row=>row[c]));
  }

  function reverseRows(a){ return a.map(row=>row.slice().reverse()); }

  function collapseRow(row){
    let values=row.filter(Boolean);
    let result=[], gained=0;
    for(let i=0;i<values.length;i++){
      if(values[i]===values[i+1]){
        const n=values[i]*2;
        result.push(n);
        gained+=n;
        i++;
      } else result.push(values[i]);
    }
    while(result.length<4) result.push(0);
    return {row:result,gained};
  }

  function move(direction){
    let before=JSON.stringify(grid);
    let work=grid.map(r=>r.slice());
    let gained=0;

    if(direction==='left'){
      work=work.map(row=>{
        const x=collapseRow(row); gained+=x.gained; return x.row;
      });
    } else if(direction==='right'){
      work=reverseRows(work).map(row=>{
        const x=collapseRow(row); gained+=x.gained; return x.row;
      });
      work=reverseRows(work);
    } else if(direction==='up'){
      work=transpose(work);
      work=work.map(row=>{
        const x=collapseRow(row); gained+=x.gained; return x.row;
      });
      work=transpose(work);
    } else if(direction==='down'){
      work=transpose(work);
      work=reverseRows(work).map(row=>{
        const x=collapseRow(row); gained+=x.gained; return x.row;
      });
      work=transpose(reverseRows(work));
    }

    if(JSON.stringify(work)!==before){
      grid=work;
      score+=gained;
      if(score>best){
        best=score;
        localStorage.setItem('2048-best',best);
      }
      addRandom();
      render();
    }
  }

  function reset(){
    grid=[
      [0,0,2,0],
      [0,0,0,0],
      [0,2,4,2],
      [0,0,8,2]
    ];
    score=20;
    render();
  }

  document.addEventListener('keydown', e=>{
    const keys={
      ArrowLeft:'left',a:'left',A:'left',
      ArrowRight:'right',d:'right',D:'right',
      ArrowUp:'up',w:'up',W:'up',
      ArrowDown:'down',s:'down',S:'down'
    };
    if(keys[e.key]){
      e.preventDefault();
      move(keys[e.key]);
    }
  });

  let startX=0,startY=0;
  boardEl.addEventListener('pointerdown',e=>{
    startX=e.clientX; startY=e.clientY;
    boardEl.setPointerCapture?.(e.pointerId);
  });
  boardEl.addEventListener('pointerup',e=>{
    const dx=e.clientX-startX, dy=e.clientY-startY;
    if(Math.max(Math.abs(dx),Math.abs(dy))<25)return;
    if(Math.abs(dx)>Math.abs(dy)) move(dx>0?'right':'left');
    else move(dy>0?'down':'up');
  });

  document.getElementById('newGame').addEventListener('click',reset);
  document.getElementById('classicBtn').addEventListener('click',()=>{
    document.getElementById('classicBtn').classList.add('active');
    document.getElementById('cupcakesBtn').classList.remove('active');
  });
  document.getElementById('cupcakesBtn').addEventListener('click',()=>{
    document.getElementById('cupcakesBtn').classList.add('active');
    document.getElementById('classicBtn').classList.remove('active');
  });
  document.getElementById('upBtn').addEventListener('click',()=>{
    window.scrollTo({top:0,behavior:'smooth'});
  });

  render();
})();
</script>
