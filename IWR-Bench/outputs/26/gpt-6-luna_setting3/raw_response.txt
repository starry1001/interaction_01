function draw(){
 const w=canvas.width/devicePixelRatio,h=canvas.height/devicePixelRatio,d=devicePixelRatio;
 ctx.setTransform(d,0,0,d,0,0);ctx.clearRect(0,0,w,h);
 let scale=75, ox=w/2, oy=h*.54;
 ctx.strokeStyle='#eeeeee';ctx.lineWidth=1;
 for(let x=ox%scale;x<w;x+=scale){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,h);ctx.stroke()}
 for(let y=oy%scale;y<h;y+=scale){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(w,y);ctx.stroke()}
 ctx.strokeStyle='#c8c8c8';ctx.lineWidth=1.3;
 for(let x=ox%scale;x<w;x+=scale){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,h);ctx.stroke()}
 for(let y=oy%scale;y<h;y+=scale){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(w,y);ctx.stroke()}
 ctx.strokeStyle='#444';ctx.lineWidth=1.5;
 ctx.beginPath();ctx.moveTo(ox,0);ctx.lineTo(ox,h);ctx.stroke();
 ctx.beginPath();ctx.moveTo(0,oy);ctx.lineTo(w,oy);ctx.stroke();
 ctx.fillStyle='#444';ctx.font='16px Arial';
 for(let i=-5;i<=5;i++){if(i!==0){ctx.fillText(i,ox+i*scale-6,oy+20);ctx.fillText(i,ox-23,oy-i*scale+5)}}
 ctx.fillText('0',ox-17,oy+22);
 if(expression && expression.replace(/\s/g,'').includes('sin(x')){
   ctx.strokeStyle='#399b82';ctx.lineWidth=3;ctx.beginPath();
   for(let px=-10;px<=w+10;px+=2){
    let x=(px-ox)/scale, y=Math.sin(x), py=oy-y*scale;
    if(px===-10)ctx.moveTo(px,py);else ctx.lineTo(px,py);
   }ctx.stroke();
   [-3.14,-1.57,0,1.57,3.14].forEach(x=>{let px=ox+x*scale,py=oy-Math.sin(x)*scale;ctx.fillStyle='#777';ctx.beginPath();ctx.arc(px,py,7,0,Math.PI*2);ctx.fill()})
 }
}
