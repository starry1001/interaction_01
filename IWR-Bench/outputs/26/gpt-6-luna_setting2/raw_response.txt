<script>
(function(){
  const graph = document.getElementById('graph');
  const svg = document.getElementById('plot');
  const expr = document.getElementById('expression');
  const ns = 'http://www.w3.org/2000/svg';

  function el(tag, attrs, parent) {
    const node = document.createElementNS(ns, tag);
    Object.keys(attrs || {}).forEach(k => node.setAttribute(k, attrs[k]));
    if (parent) parent.appendChild(node);
    return node;
  }
  function draw() {
    const w = graph.clientWidth, h = graph.clientHeight;
    if (!w || !h) return;
    svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
    svg.replaceChildren();
    const unit = w / 8.5, ox = w / 2, oy = h / 2;

    el('line', {x1:0,y1:oy,x2:w-5,y2:oy,class:'axis'}, svg);
    el('path', {d:`M ${w-9} ${oy-4} L ${w-4} ${oy} L ${w-9} ${oy+4}`,class:'axis-arrow'}, svg);
    el('line', {x1:ox,y1:0,x2:ox,y2:h,class:'axis'}, svg);
    el('path', {d:`M ${ox-4} 6 L ${ox} 1 L ${ox+4} 6`,class:'axis-arrow'}, svg);

    for (let i=-4; i<=4; i++) {
      if (i !== 0) {
        const t = el('text',{x:ox+i*unit,y:oy+17,class:'axis-label'},svg);
        t.textContent = i;
        const ty = el('text',{x:ox-13,y:oy-i*unit+4,class:'axis-label','text-anchor':'end'},svg);
        ty.textContent = i;
      } else {
        const zero = el('text',{x:ox-10,y:oy+16,class:'axis-label','text-anchor':'end'},svg);
        zero.textContent='0';
      }
    }

    let d = '';
    const xStart = 0.055, xEnd = Math.min(4.5, (w-ox)/unit);
    const steps = 500;
    for (let i=0;i<=steps;i++) {
      const x = xStart + (xEnd-xStart)*i/steps;
      const y = Math.sin(x*Math.log(x));
      const px = ox+x*unit, py = oy-y*unit;
      d += (i ? ' L ' : 'M ') + px.toFixed(2) + ' ' + py.toFixed(2);
    }
    el('path',{d,class:'curve'},svg);
    [[1,0],[Math.exp(Math.sqrt(Math.PI/2)),1],[Math.exp(Math.sqrt(3*Math.PI/2)),-1],[Math.exp(Math.sqrt(Math.PI)) ,0],[Math.exp(Math.sqrt(2*Math.PI)),0]].forEach(p=>{
      const x=p[0], y=p[1];
      if(x < xEnd+.2) el('circle',{cx:ox+x*unit,cy:oy-y*unit,r:5.4,class:'curve-point'},svg);
    });
  }
  draw();
  window.addEventListener('resize', draw);

  document.querySelectorAll('.tab[data-layout]').forEach(tab=>{
    tab.addEventListener('click',()=>{
      document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));
      tab.classList.add('active');
      const numeric = document.getElementById('numeric');
      const letters = document.getElementById('letters');
      const layout = tab.dataset.layout;
      numeric.style.display = layout === 'numeric' || layout === 'function' || layout === 'symbols' ? 'flex' : 'none';
      letters.style.display = layout === 'letters' ? 'flex' : 'none';
      if(layout === 'function' || layout === 'symbols') {
        const keys = numeric.querySelectorAll('.key');
        if(layout === 'function') {
          const vals=['sin','cos','tan','%','!','$','°','sin⁻¹','cos⁻¹','tan⁻¹','{','}','≤','≥','ln','log₁₀','logₐ','d/dx','∫','i','⌫','e','10ˣ','√','∷','‹','›','↵'];
          keys.forEach((k,i)=>{ if(vals[i]) k.textContent=vals[i]; });
        } else {
          const vals=['!','@','#','$','%','^','&','*','(',')','-','_','=','+','[',']','{','}','\\','|',';',':','"',"'",'<','>','?','↵'];
          keys.forEach((k,i)=>{ if(vals[i]) k.textContent=vals[i]; });
        }
      }
    });
  });

  document.querySelectorAll('.key').forEach(key=>{
    key.addEventListener('click',()=>{
      const action=key.dataset.action;
      if(action==='backspace') expr.textContent=expr.textContent.slice(0,-1);
      else if(action==='enter') return;
      else {
        const v=key.textContent.trim();
        if(v && v!=='⌃' && v!=='‹' && v!=='›' && v!=='αβγ') expr.textContent += v;
      }
    });
  });
  document.querySelector('.close-keyboard').addEventListener('click',()=>{
    document.getElementById('keyboard').style.display='none';
  });
  document.querySelector('.entry-plus').addEventListener('click',()=>{
    expr.textContent='';
    document.getElementById('keyboard').style.display='flex';
  });
})();
</script>
