<script>
(() => {
  const app = document.getElementById('app');
  const clock = document.getElementById('clock');
  const timerButton = document.getElementById('timerButton');
  const skipButton = document.getElementById('skipButton');
  const sessionText = document.getElementById('sessionText');
  const editor = document.getElementById('editor');
  const taskInput = document.getElementById('taskInput');
  const estimate = document.getElementById('estimate');
  const taskList = document.getElementById('taskList');
  const modes = [...document.querySelectorAll('.mode')];
  const lengths = { pomodoro: 25, short: 5, long: 15 };
  let mode = 'pomodoro', seconds = 25 * 60, running = false, ticker = null;
  let tasks = [], editIndex = -1, pomodoros = 0, session = 1;

  function setColor(selected) {
    const colors = { pomodoro:'#b13b4d', short:'#856064', long:'#408b8d' };
    app.style.setProperty('--primary', colors[selected]);
    document.documentElement.style.background = colors[selected];
    document.body.style.background = colors[selected];
  }
  function renderClock() {
    clock.textContent = String(Math.floor(seconds / 60)).padStart(2,'0') + ':' + String(seconds % 60).padStart(2,'0');
  }
  function chooseMode(next) {
    if (running) stopTimer();
    mode = next;
    seconds = lengths[mode] * 60;
    modes.forEach(button => button.classList.toggle('active', button.dataset.mode === mode));
    setColor(mode);
    sessionText.textContent = mode === 'pomodoro' ? 'Time to focus!' : 'Time for a break!';
    renderClock();
  }
  function stopTimer() {
    running = false;
    clearInterval(ticker);
    timerButton.textContent = 'START';
    skipButton.classList.remove('show');
  }
  function completeTimer() {
    stopTimer();
    if (mode === 'pomodoro') {
      pomodoros++;
      session++;
      sessionText.textContent = 'Time for a break!';
      chooseMode(pomodoros % 4 === 0 ? 'long' : 'short');
    } else {
      sessionText.textContent = 'Time to focus!';
      chooseMode('pomodoro');
    }
    document.querySelector('.session strong').textContent = '#' + session;
  }
  modes.forEach(button => button.addEventListener('click', () => chooseMode(button.dataset.mode)));
  timerButton.addEventListener('click', () => {
    if (!running) {
      running = true;
      timerButton.textContent = 'PAUSE';
      skipButton.classList.add('show');
      ticker = setInterval(() => {
        if (seconds > 0) { seconds--; renderClock(); }
        else completeTimer();
      }, 1000);
    } else stopTimer();
  });
  skipButton.addEventListener('click', completeTimer);

  function openEditor(index = -1) {
    editIndex = index;
    editor.classList.add('open');
    if (index >= 0) {
      taskInput.value = tasks[index].title;
      estimate.value = tasks[index].estimate;
    } else {
      taskInput.value = '';
      estimate.value = 1;
    }
    document.getElementById('addTaskButton').style.display = 'none';
    taskInput.focus();
  }
  function closeEditor() {
    editor.classList.remove('open');
    document.getElementById('addTaskButton').style.display = '';
    editIndex = -1;
  }
  document.getElementById('addTaskButton').addEventListener('click', () => openEditor());
  document.getElementById('cancelEdit').addEventListener('click', closeEditor);
  document.getElementById('saveTask').addEventListener('click', () => {
    const title = taskInput.value.trim();
    if (!title) { taskInput.focus(); return; }
    const item = { title, estimate: Math.max(1, parseInt(estimate.value,10) || 1), done:false };
    if (editIndex >= 0) tasks[editIndex] = item;
    else tasks.push(item);
    closeEditor();
    renderTasks();
  });
  document.getElementById('estUp').addEventListener('click', () => estimate.value = (+estimate.value || 1) + 1);
  document.getElementById('estDown').addEventListener('click', () => estimate.value = Math.max(1, (+estimate.value || 1) - 1));
  taskInput.addEventListener('keydown', e => { if (e.key === 'Enter') document.getElementById('saveTask').click(); });

  function renderTasks() {
    taskList.innerHTML = '';
    tasks.forEach((task, index) => {
      const row = document.createElement('div');
      row.className = 'task-row' + (task.done ? ' done' : '');
      const check = document.createElement('button');
      check.className = 'complete btn';
      check.textContent = task.done ? '✓' : '✓';
      check.setAttribute('aria-label','Complete task');
      check.addEventListener('click', () => { task.done = !task.done; renderTasks(); });
      const title = document.createElement('span');
      title.className = 'task-title';
      title.textContent = task.title;
      title.addEventListener('click', () => openEditor(index));
      const count = document.createElement('span');
      count.className = 'task-count';
      count.textContent = '0 / ' + task.estimate;
      const options = document.createElement('button');
      options.className = 'task-options btn';
      options.textContent = '⋮';
      options.setAttribute('aria-label','Edit task');
      options.addEventListener('click', () => openEditor(index));
      row.append(check, title, count, options);
      taskList.appendChild(row);
    });
    const total = tasks.reduce((sum,t) => sum + t.estimate, 0);
    document.getElementById('pomCount').textContent = pomodoros + ' / ' + total;
    document.getElementById('toast').classList.toggle('show', tasks.length > 0 && tasks.every(t => t.done));
  }
  document.getElementById('moreButton').addEventListener('click', () => document.getElementById('taskMenu').classList.toggle('open'));
  document.getElementById('clearDone').addEventListener('click', () => {
    tasks = tasks.filter(task => !task.done);
    renderTasks();
    document.getElementById('taskMenu').classList.remove('open');
  });
  document.getElementById('clearAll').addEventListener('click', () => {
    tasks = []; renderTasks(); document.getElementById('taskMenu').classList.remove('open');
  });
  document.getElementById('addRoutine').addEventListener('click', () => {
    tasks.push({title:'My routine task', estimate:1, done:false});
    renderTasks(); document.getElementById('taskMenu').classList.remove('open');
  });
  document.getElementById('clearToast').addEventListener('click', () => { tasks=[]; renderTasks(); });
})();
</script>
