<script>
(() => {
  const app = document.getElementById('app');
  const timeEl = document.getElementById('time');
  const start = document.getElementById('start');
  const modes = [...document.querySelectorAll('.mode')];
  const sessionName = document.getElementById('sessionName');
  const editor = document.getElementById('editor');
  const input = document.getElementById('taskInput');
  const estimate = document.getElementById('estimate');
  const addTask = document.getElementById('addTask');
  const taskList = document.getElementById('taskList');
  const summary = document.getElementById('summary');
  const finished = document.getElementById('finished');
  let selected = 'pomodoro', running = false, seconds = 1500, ticker = null;
  let tasks = [];

  const durations = { pomodoro: 25, short: 5, long: 15 };
  const sessionLabels = { pomodoro: 'Time to focus!', short: 'Time for a break!', long: 'Time for a break!' };
  function format() {
    timeEl.textContent = `${String(Math.floor(seconds / 60)).padStart(2,'0')}:${String(seconds % 60).padStart(2,'0')}`;
  }
  function chooseMode(mode) {
    selected = mode;
    modes.forEach(b => b.classList.toggle('active', b.dataset.mode === mode));
    app.classList.toggle('break', mode !== 'pomodoro');
    seconds = durations[mode] * 60;
    sessionName.textContent = sessionLabels[mode];
    stopTimer();
    format();
  }
  function stopTimer() {
    running = false;
    clearInterval(ticker);
    app.classList.remove('running');
    start.textContent = 'START';
  }
  modes.forEach(b => b.addEventListener('click', () => chooseMode(b.dataset.mode)));
  start.addEventListener('click', () => {
    if (running) { stopTimer(); return; }
    running = true;
    app.classList.add('running');
    start.textContent = 'PAUSE';
    ticker = setInterval(() => {
      if (seconds > 0) { seconds--; format(); }
      else stopTimer();
    }, 1000);
  });
  document.getElementById('skip').addEventListener('click', () => {
    stopTimer();
    seconds = durations[selected] * 60;
    format();
  });

  function openEditor() {
    editor.classList.add('open');
    addTask.style.display = 'none';
    input.value = '';
    estimate.value = 1;
    setTimeout(() => input.focus(), 30);
  }
  function closeEditor() {
    editor.classList.remove('open');
    addTask.style.display = '';
  }
  addTask.addEventListener('click', openEditor);
  document.getElementById('cancel').addEventListener('click', closeEditor);
  document.getElementById('up').addEventListener('click', () => estimate.value = Math.max(1, (+estimate.value || 1) + 1));
  document.getElementById('down').addEventListener('click', () => estimate.value = Math.max(1, (+estimate.value || 1) - 1));

  function updateSummary() {
    if (!tasks.length) {
      summary.style.display = 'none';
      finished.classList.remove('show');
      return;
    }
    summary.style.display = 'flex';
    const total = tasks.reduce((n, t) => n + t.est, 0);
    const done = tasks.filter(t => t.done).length;
    document.getElementById('pomos').textContent = `${done} / ${total}`;
    finished.classList.toggle('show', tasks.every(t => t.done));
  }
  function renderTasks() {
    taskList.innerHTML = '';
    tasks.forEach((task, i) => {
      const row = document.createElement('div');
      row.className = 'task-row' + (task.done ? ' done' : '');
      row.innerHTML = `<button class="btn task-check" aria-label="Complete task">${task.done ? '✓' : '✓'}</button>
        <span class="task-title"></span><span class="task-count">0 / ${task.est}</span>
        <button class="btn task-menu" aria-label="Task options">⋮</button>`;
      row.querySelector('.task-title').textContent = task.title;
      row.querySelector('.task-check').addEventListener('click', () => {
        task.done = !task.done; renderTasks(); updateSummary();
      });
      row.querySelector('.task-menu').addEventListener('click', () => {
        if (confirm('Delete this task?')) { tasks.splice(i, 1); renderTasks(); updateSummary(); }
      });
      taskList.appendChild(row);
    });
    taskList.appendChild(addTask);
    addTask.style.display = editor.classList.contains('open') ? 'none' : '';
  }
  document.getElementById('save').addEventListener('click', () => {
    const value = input.value.trim();
    if (!value) { input.focus(); return; }
    tasks.push({ title: value, est: Math.max(1, +estimate.value || 1), done: false });
    renderTasks();
    updateSummary();
    openEditor();
  });
  document.getElementById('clear').addEventListener('click', () => {
    tasks = [];
    renderTasks();
    updateSummary();
  });
})();
</script>
</body>
</html>
