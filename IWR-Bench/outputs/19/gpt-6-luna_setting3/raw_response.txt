function filterComments() {
  const type = document.getElementById("typeFilter").value.toLowerCase();
  const author = document.getElementById("authorFilter").value.toLowerCase();
  const key = document.getElementById("keyword").value.toLowerCase();
  document.querySelectorAll(".comment-card").forEach(c=>{
    const text = c.innerText.toLowerCase();
    c.classList.toggle("hidden", (type && !text.includes(type.toLowerCase())) || (author && !text.includes(author)) || (key && !text.includes(key)));
  });
}

document.addEventListener("click", e=>{
  const go = e.target.closest("[data-go]");
  if(go) {
    e.preventDefault();
    const dest=go.dataset.go;
    if(dest==="home") page="home";
    else if(dest==="years") page="years";
    else if(dest==="year") page="year";
    else if(dest==="track") page="track";
    else if(dest==="paper") page="paper";
    else if(dest==="allvenues") { document.getElementById("allVenues")?.scrollIntoView(); return; }
    render(); window.scrollTo(0,0); return;
  }
  const detail = e.target.closest("[data-detail]");
  if(detail) {
    e.preventDefault();
    const el=document.getElementById("detail"+detail.dataset.detail);
    el.classList.toggle("open"); detail.textContent=el.classList.contains("open")?"Hide details":"Show details";
    return;
  }
  const modal = e.target.closest("[data-modal]");
  if(modal) { e.preventDefault(); showModal(modal.dataset.modal); return; }
  if(e.target.closest("[data-close]")) document.getElementById("modalShade").classList.remove("show");
  if(e.target.classList.contains("tab")) {
    document.querySelectorAll(".tab").forEach(t=>t.classList.remove("active"));
    e.target.classList.add("active");
  }
});
document.getElementById("modalShade").addEventListener("click",e=>{if(e.target.id==="modalShade") e.currentTarget.classList.remove("show");});
render();
