const questionSets = [
  [
    "You regularly make new friends.",
    "Complex and novel ideas excite you more than simple and straightforward ones.",
    "You usually feel more persuaded by what resonates emotionally with you than by factual arguments.",
    "Your living and working spaces are clean and organized.",

  ],
  [
    "You prioritize and plan tasks effectively, often completing them well before the deadline.",
    "People’s stories and emotions speak louder to you than numbers or data.",
    "You like to use organizing tools like schedules and lists.",
    "Even a small mistake can cause you to doubt your overall abilities and knowledge.",
    "You feel comfortable approaching someone you find interesting and striking up a conversation.",
    "You are not too interested in discussing various interpretations and analyses of creative works."
  ],
  [
    "You prefer to completely finish one project before starting another.",
    "You enjoy lively discussions about ethical dilemmas.",
    "You generally prefer to act immediately rather than spend time planning.",
    "You are a relatively reserved and quiet person.",
    "You are drawn to various forms of creative expression."
  ]
];
let currentStep = 0;
const answers = {};
function startTest() {
  document.body.classList.add("testing");
  currentStep = 0;
  renderQuestions();
  window.scrollTo({top:0, behavior:"smooth"});
}
function renderQuestions() {
  const holder = document.getElementById("questions");
  holder.innerHTML = "";
  const qlist = questionSets[Math.min(currentStep, questionSets.length - 1)];
  qlist.forEach((text, index) => {
    const key = currentStep + "-" + index;
    const row = document.createElement("article");
    row.className = "question" + ((index === 0 || answers[key] !== undefined) ? " active" : "");
    row.innerHTML = `<h3>${text}</h3><div class="answers">
      <span class="agree-label">Agree</span>
      ${[0,1,2,3,4,5,6].map((v) => {
        const selected = answers[key] === v;
        const purple = v >= 4;
        return `<button class="btn choice ${selected ? "selected" : ""} ${purple ? "purple" : ""}" aria-label="Response ${v+1}" onclick="selectAnswer('${key}',${v},this)"><span class="dot"></span></button>`;
      }).join("")}
      <span class="disagree-label">Disagree</span>
    </div>`;
    holder.appendChild(row);
  });
  const progress = document.getElementById("progress");
  progress.style.display = currentStep === 0 ? "none" : "flex";
  document.getElementById("stepCount").textContent = `Step ${currentStep + 1} of 10`;
  document.getElementById("barFill").style.width = `${(currentStep + 1) * 10}%`;
  document.getElementById("nextRow").style.display = "block";
}
function selectAnswer(key, value, button) {
  answers[key] = value;
  const row = button.closest(".question");
  row.classList.add("active", "answered");
  row.querySelectorAll(".choice").forEach(c => c.classList.remove("selected"));
  button.classList.add("selected");
}
function nextStep() {
  currentStep++;
  renderQuestions();
  window.scrollTo({top: document.querySelector(".test-area").offsetTop - 110, behavior:"smooth"});
}
function previousStep() {
  currentStep = Math.max(0, currentStep - 1);
  renderQuestions();
  window.scrollTo({top: document.querySelector(".test-area").offsetTop - 110, behavior:"smooth"});
}
renderQuestions();
</script>
