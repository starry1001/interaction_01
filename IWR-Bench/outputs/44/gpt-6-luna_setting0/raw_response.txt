<script>
  const red = document.getElementById('redHighlight');
  const redEye = document.getElementById('redEye');
  const statsBtn = document.getElementById('statsBtn');
  const lessBtn = document.getElementById('lessBtn');
  const expanded = document.getElementById('expandedStats');
  const wordLabel = document.getElementById('wordLabel');
  const menuBtn = document.getElementById('menuBtn');
  const dropdown = document.getElementById('dropdown');
  const modal = document.getElementById('modal');

  redEye.addEventListener('click', e => {
    e.stopPropagation();
    red.classList.toggle('hidden-highlight');
  });
  statsBtn.addEventListener('click', () => {
    expanded.style.display = 'block';
    statsBtn.style.display = 'none';
    wordLabel.style.display = 'none';
  });
  lessBtn.addEventListener('click', () => {
    expanded.style.display = 'none';
    statsBtn.style.display = 'inline-block';
    wordLabel.style.display = 'block';
  });
  menuBtn.addEventListener('click', e => {
    e.stopPropagation();
    dropdown.style.display = dropdown.style.display === 'block' ? 'none' : 'block';
  });
  document.addEventListener('click', () => dropdown.style.display='none');
  document.getElementById('helpBtn').addEventListener('click', () => modal.style.display='block');
  document.getElementById('tutorialBtn').addEventListener('click', () => modal.style.display='block');
  document.getElementById('closeModal').addEventListener('click', () => modal.style.display='none');
  modal.addEventListener('click', e => { if(e.target === modal) modal.style.display='none'; });
  document.querySelectorAll('.tab').forEach(tab => {
    tab.addEventListener('click', () => {
      document.querySelectorAll('.tab').forEach(x => x.classList.remove('active'));
      tab.classList.add('active');
      if(tab.textContent.trim() === 'Write'){
        document.querySelectorAll('.yellow,.red,.purple,.blue').forEach(x => x.classList.add('hidden-highlight'));
      } else {
        document.querySelectorAll('.yellow,.red,.purple,.blue').forEach(x => x.classList.remove('hidden-highlight'));
      }
    });
  });
</script>
