<script>
  const prices = {
    USD: {
      annual: ['$2.99', '$4.99', '$19.95', '$7.99'],
      monthly: ['$3.99', '$6.95', '$24.95', '$9.99']
    },
    CAD: {
      annual: ['$3.99', '$6.95', '$24.95', '$9.99'],
      monthly: ['$4.99', '$7.95', '$29.95', '$11.99']
    },
    EUR: {
      annual: ['€2.99', '€4.99', '€17.95', '€6.99'],
      monthly: ['€3.39', '€5.95', '€20.95', '€8.99']
    }
  };
  const flags = { USD: '🇺🇸', CAD: '🇨🇦', EUR: '🇪🇺' };
  let currency = 'USD';
  let billing = 'annual';

  function updatePrices() {
    const values = prices[currency][billing];
    document.querySelectorAll('.price').forEach((el, i) => el.textContent = values[i]);
    document.querySelectorAll('.currency-label').forEach(el => el.textContent = currency);
    document.querySelectorAll('.billing-note').forEach((el, i) => {
      const regular = [
        billing === 'annual' ? 'Per month. Paid annually.' : 'Per month.',
        billing === 'annual' ? 'Per month. Paid annually.' : 'Per month.',
        billing === 'annual' ? 'Up to 10 users/month. Paid annually.' : 'Up to 10 users per month.',
        billing === 'annual' ? 'Per user, per month. Paid annually.' : 'Per user, per month.'
      ];
      el.textContent = regular[i];
    });
  }
  document.querySelectorAll('input[name="billing"]').forEach(input => {
    input.addEventListener('change', () => {
      billing = input.value;
      updatePrices();
    });
  });

  const toggle = document.getElementById('currencyToggle');
  const menu = document.getElementById('currencyMenu');
  toggle.addEventListener('click', () => {
    const open = menu.hidden;
    menu.hidden = !open;
    toggle.setAttribute('aria-expanded', String(open));
  });
  document.querySelectorAll('.currency-option').forEach(option => {
    option.addEventListener('click', () => {
      currency = option.dataset.currency;
      document.getElementById('currentCode').textContent = currency;
      document.getElementById('currentFlag').textContent = flags[currency];
      document.querySelectorAll('.currency-option').forEach(item => {
        const selected = item.dataset.currency === currency;
        item.classList.toggle('selected', selected);
        item.querySelector('.tick').textContent = selected ? '✓' : '';
      });
      menu.hidden = true;
      toggle.setAttribute('aria-expanded', 'false');
      updatePrices();
    });
  });
  document.querySelector('.promo-close').addEventListener('click', () => {
    document.getElementById('promo').style.display = 'none';
  });
  document.querySelectorAll('.trial, .signin, .nav-link, .plan-switch button, .promo a, .brand').forEach(el => {
    el.addEventListener('click', event => {
      if (el.tagName === 'A') event.preventDefault();
    });
  });
</script>
