<script>
  const prices = {
    annual: {
      USD: ['$2.99', '$4.99', '$19.95', '$7.99'],
      CAD: ['$3.99', '$5.99', '$24.95', '$9.99'],
      EUR: ['€2.99', '€4.99', '€19.95', '€7.99']
    },
    monthly: {
      USD: ['$3.99', '$6.95', '$24.95', '$9.99'],
      CAD: ['$4.99', '$7.95', '$29.95', '$11.99'],
      EUR: ['€3.39', '€5.95', '€20.95', '€8.99']
    }
  };
  const planNames = ['individual', 'families', 'teams', 'business'];
  const currencies = { USD: '🇺🇸', CAD: '🇨🇦', EUR: '🇪🇺' };
  let billingType = 'annual';
  let currency = 'USD';

  function updatePrices() {
    document.querySelectorAll('.price').forEach((el, index) => {
      el.textContent = prices[billingType][currency][index];
    });
    document.querySelectorAll('.currency-code').forEach(el => el.textContent = currency);
    document.querySelectorAll('.price-note').forEach((el, index) => {
      if (billingType === 'monthly') {
        el.textContent = ['Per month.', 'Per month.', 'Up to 10 users per month.', 'Per user, per month.'][index];
      } else {
        el.textContent = ['Per month. Paid annually.', 'Per month. Paid annually.', 'Up to 10 users/month. Paid annually.', 'Per user, per month. Paid annually.'][index];
      }
    });
  }

  document.querySelectorAll('input[name="billing"]').forEach(input => {
    input.addEventListener('change', () => {
      billingType = input.value;
      updatePrices();
    });
  });

  const trigger = document.getElementById('currencyTrigger');
  const menu = document.getElementById('currencyMenu');
  trigger.addEventListener('click', () => {
    const open = menu.classList.toggle('open');
    trigger.setAttribute('aria-expanded', open ? 'true' : 'false');
  });

  document.querySelectorAll('.currency-option').forEach(option => {
    option.addEventListener('click', () => {
      currency = option.dataset.currency;
      document.getElementById('currentCode').textContent = currency;
      document.getElementById('currentFlag').textContent = currencies[currency];
      document.querySelectorAll('.currency-option').forEach(item => {
        item.classList.toggle('selected', item.dataset.currency === currency);
        item.querySelector('.check').textContent = item.dataset.currency === currency ? '✓' : '';
      });
      menu.classList.remove('open');
      trigger.setAttribute('aria-expanded', 'false');
      updatePrices();
    });
  });

  document.addEventListener('click', event => {
    if (!event.target.closest('.currency-wrap')) {
      menu.classList.remove('open');
      trigger.setAttribute('aria-expanded', 'false');
    }
  });

  document.getElementById('infoButton').addEventListener('click', () => {
    document.getElementById('infoTooltip').classList.toggle('show');
  });

  function selectTab(button) {
    document.querySelectorAll('.tab').forEach(tab => {
      tab.classList.remove('active');
      tab.setAttribute('aria-selected', 'false');
    });
    button.classList.add('active');
    button.setAttribute('aria-selected', 'true');
  }

  document.querySelectorAll('.trial, .sign-in, .nav-link').forEach(button => {
    button.addEventListener('click', event => event.preventDefault());
  });

  document.querySelectorAll('.currency-option').forEach(item => {
    item.classList.toggle('selected', item.dataset.currency === currency);
    item.querySelector('.check').textContent = item.dataset.currency === currency ? '✓' : '';
  });
  updatePrices();
</script>
