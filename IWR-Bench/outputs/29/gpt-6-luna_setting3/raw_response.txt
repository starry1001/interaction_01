<script>
  const prices = {
    USD: {flag:"🇺🇸", symbols:["$","$","$","$"], vals:["3.99","6.95","24.95","9.99"]},
    CAD: {flag:"🇨🇦", symbols:["$","$","$","$"], vals:["4.99","7.95","29.95","11.99"]},
    EUR: {flag:"🇪🇺", symbols:["€","€","€","€"], vals:["3.39","5.95","20.95","8.99"]}
  };
  const plans = [
    ["Individual","Take control of your online security.","Per month."],
    ["Families","Peace of mind for your entire family.","Per month."],
    ["Teams Starter Pack","Secure your small team.","Up to 10 users per month."],
    ["Business","Security that adapts to your business.","Per user, per month."]
  ];
  let monthly = true, currency = "USD", dropdownOpen = false;

  function renderCards(){
    const d = prices[currency];
    document.getElementById("cards").innerHTML = plans.map((p,i)=>`
      <article class="card">
        <h2>${p[0]}</h2>
        <div class="desc">${p[1]}</div>
        <div class="price">${d.symbols[i]}${d.vals[i]}<small>${currency}</small></div>
        <div class="period">${monthly ? p[2] : (i===2 ? "Up to 10 users/month. Paid annually." : i===3 ? "Per user, per month. Paid annually." : "Per month. Paid annually.")}</div>
        <button class="trial btn">Try FREE for 14 days</button>
      </article>`).join("");
  }
  function updateBilling(){
    document.getElementById("annualRadio").classList.toggle("selected",!monthly);
    document.getElementById("monthlyRadio").classList.toggle("selected",monthly);
    renderCards();
  }
  function setCurrency(c){
    currency=c;
    document.getElementById("currencyFlag").textContent=prices[c].flag;
    document.getElementById("currencyCode").textContent=c;
    document.querySelectorAll(".drop-option").forEach(x=>{
      x.classList.toggle("selected",x.dataset.currency===c);
      x.querySelector(".check").textContent=x.dataset.currency===c?"✓":"";
    });
    renderCards();
  }
  document.querySelector(".banner-close").onclick=()=>document.getElementById("banner").style.display="none";
  document.getElementById("annualLabel").onclick=()=>{monthly=false;updateBilling()};
  document.getElementById("monthlyLabel").onclick=()=>{monthly=true;updateBilling()};
  document.getElementById("currencyButton").onclick=()=>{
    dropdownOpen=!dropdownOpen;
    document.getElementById("dropdown").style.display=dropdownOpen?"block":"none";
    document.getElementById("currencyArrow").textContent=dropdownOpen?"⌃":"⌄";
  };
  document.querySelectorAll(".drop-option").forEach(o=>o.onclick=()=>{
    setCurrency(o.dataset.currency);dropdownOpen=false;
    document.getElementById("dropdown").style.display="none";
    document.getElementById("currencyArrow").textContent="⌄";
  });
  document.addEventListener("click",e=>{
    if(!e.target.closest(".currency-wrap") && dropdownOpen){
      dropdownOpen=false;document.getElementById("dropdown").style.display="none";
      document.getElementById("currencyArrow").textContent="⌄";
    }
  });
  setCurrency("USD");
  updateBilling();
</script>
