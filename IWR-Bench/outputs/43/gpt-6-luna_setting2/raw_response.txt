<script>
  const qtyOutput = document.getElementById('qty');
  let quantity = 1;
  document.getElementById('plus').addEventListener('click', () => {
    quantity = Math.min(99, quantity + 1);
    qtyOutput.textContent = quantity;
  });
  document.getElementById('minus').addEventListener('click', () => {
    quantity = Math.max(1, quantity - 1);
    qtyOutput.textContent = quantity;
  });

  document.querySelectorAll('.delivery-box').forEach(box => {
    box.addEventListener('click', () => {
      document.querySelectorAll('.delivery-box').forEach(item => item.classList.remove('selected'));
      box.classList.add('selected');
    });
  });

  document.querySelectorAll('.swatch').forEach(swatch => {
    swatch.addEventListener('click', () => {
      document.querySelectorAll('.swatch').forEach(item => item.classList.remove('selected'));
      swatch.classList.add('selected');
      document.getElementById('colorName').textContent = swatch.dataset.color;
    });
  });

  document.querySelectorAll('.thumb').forEach(thumb => {
    thumb.addEventListener('click', () => {
      document.querySelectorAll('.thumb').forEach(item => item.classList.remove('active'));
      thumb.classList.add('active');
      document.getElementById('heroImage').src = '__PLACEHOLDER_ASSETS_BASE_DIR__/' + thumb.dataset.image;
    });
  });

  document.getElementById('detailsLink').addEventListener('click', () => {
    const extra = document.getElementById('extraDetails');
    extra.style.display = extra.style.display === 'none' ? 'block' : 'none';
  });

  document.getElementById('addCart').addEventListener('click', () => {
    const toast = document.getElementById('toast');
    toast.textContent = quantity + (quantity === 1 ? ' item added to cart' : ' items added to cart');
    toast.style.display = 'block';
    setTimeout(() => toast.style.display = 'none', 2300);
  });

  document.querySelectorAll('.heart, .item-heart').forEach(heart => {
    heart.addEventListener('click', () => heart.textContent = heart.textContent === '♡' ? '♥' : '♡');
    heart.classList.add('btn');
  });
</script>
