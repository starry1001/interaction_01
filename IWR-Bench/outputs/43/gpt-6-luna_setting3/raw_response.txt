    });

    card.addEventListener('click',()=>{
      document.querySelectorAll('.ful-card').forEach(c=>c.classList.remove('active'));
      card.classList.add('active');
    });
  });
  document.querySelectorAll('.radio-row').forEach(row=>{
    row.addEventListener('click',()=>{
      document.querySelectorAll('.radio-row .radio').forEach(r=>r.classList.remove('selected'));
      row.querySelector('.radio').classList.add('selected');
    });
  });
  const qty=document.getElementById('qty'), monthly=document.getElementById('monthly');
  function update(){
    let n=+qty.value;
    monthly.textContent='As low as $'+(60.29*n).toFixed(2)+'/mo with PayPal';
  }
  document.getElementById('plus').onclick=()=>{qty.value=+qty.value+1;update()};
