    document.body.classList.add('coffee-mode');
  }else{
    document.body.classList.remove('coffee-mode');
  }
});

  input.value='';
  document.body.classList.remove('coffee-mode');
  input.focus();
});
input.addEventListener('input',()=>{
  clear.style.display=input.value?'block':'none';
});
