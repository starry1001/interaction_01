(() => {
  const destinationPopup=document.getElementById('destinationPopup');

  document.getElementById('destButton').addEventListener('click',e=>{
    e.stopPropagation(); const open=!destinationPopup.classList.contains('open');closePopups(destinationPopup);destinationPopup.classList.toggle('open',open);
  });
  document.getElementById('dateButton').addEventListener('click',e=>{
    e.stopPropagation(); const open=!calendarPopup.classList.contains('open');closePopups(calendarPopup);calendarPopup.classList.toggle('open',open);
  });
  document.getElementById('guestButton').addEventListener('click',e=>{
    e.stopPropagation(); const open=!guestsPopup.classList.contains('open');closePopups(guestsPopup);guestsPopup.classList.toggle('open',open);
  });
  document.querySelectorAll('.destination-row').forEach(row=>row.addEventListener('click',()=>{
    destText.textContent=row.dataset.place;clearDest.style.display='inline-block';destinationPopup.classList.remove('open');

  clearDest.addEventListener('click',e=>{e.stopPropagation();destText.textContent='Where are you going?';clearDest.style.display='none'});
  let adults=2,children=0,rooms=1;
  function refreshGuests(){document.getElementById('adultCount').textContent=adults;document.getElementById('childCount').textContent=children;document.getElementById('roomCount').textContent=rooms;guestText.textContent=`${adults} adults · ${children} children · ${rooms} room${rooms===1?'':'s'}`}
  document.querySelectorAll('[data-counter]').forEach(b=>b.addEventListener('click',e=>{
    e.stopPropagation();const key=b.dataset.counter,delta=Number(b.dataset.delta);
    if(key==='adults')adults=Math.max(1,adults+delta);
    if(key==='children')children=Math.max(0,children+delta);
    if(key==='rooms')rooms=Math.max(1,rooms+delta);
    refreshGuests();
  }));
  document.getElementById('petSwitch').addEventListener('click',e=>e.currentTarget.classList.toggle('on'));

  let year=2025,month=7,checkIn=null,checkOut=null;
  const names=['January','February','March','April','May','June','July','August','September','October','November','December'];
  const weekdays=['Su','Mo','Tu','We','Th','Fr','Sa'];
  function monthMarkup(y,m,side){
    const first=new Date(y,m,1).getDay(),total=new Date(y,m+1,0).getDate(),prevTotal=new Date(y,m,0).getDate();
    let html='<div class="month"><div class="weekdays">'+weekdays.map(w=>`<span>${w}</span>`).join('')+'</div><div class="days">';
    for(let i=0;i<42;i++){
      let n,cl='day',date;
      if(i<first){n=prevTotal-first+i+1;cl+=' muted';date=new Date(y,m-1,n)}
      else if(i>=first+total){n=i-first-total+1;cl+=' muted';date=new Date(y,m+1,n)}
      else {n=i-first+1;date=new Date(y,m,n)}
      const key=`${date.getFullYear()}-${date.getMonth()}-${date.getDate()}`;
      const inKey=checkIn&&`${checkIn.getFullYear()}-${checkIn.getMonth()}-${checkIn.getDate()}`;
      const outKey=checkOut&&`${checkOut.getFullYear()}-${checkOut.getMonth()}-${checkOut.getDate()}`;
      if(key===inKey||key===outKey)cl+=' chosen';
      else if(checkIn&&checkOut&&date>checkIn&&date<checkOut)cl+=' range';
      html+=`<button class="${cl} btn" data-date="${date.toISOString()}">${n}</button>`;
    }
    return html+'</div></div>';
  }
  function drawCalendar(){
    let y2=year,m2=month+1;if(m2>11){m2=0;y2++}

    document.getElementById('monthTitle2').textContent=`${names[m2]} ${y2}`;
    document.getElementById('calendarMonths').innerHTML=monthMarkup(year,month,0)+monthMarkup(y2,m2,1);

      const d=new Date(b.dataset.date+'');
      if(!checkIn||checkOut||d<checkIn){checkIn=d;checkOut=null}
      else checkOut=d;
      drawCalendar();
      if(checkIn&&!checkOut)dateText.textContent=`${checkIn.toLocaleDateString('en-US',{weekday:'short',month:'short',day:'numeric'})} — Check-out date`;
      if(checkIn&&checkOut){
        const a=checkIn.toLocaleDateString('en-US',{weekday:'short',month:'short',day:'numeric'});
        const z=checkOut.toLocaleDateString('en-US',{weekday:'short',month:'short',day:'numeric'});
        dateText.textContent=`${a} — ${z}`;
      }
    }));
  }
  document.getElementById('prevMonth').addEventListener('click',()=>{month--;if(month<0){month=11;year--}drawCalendar()});
  document.getElementById('nextMonth').addEventListener('click',()=>{month++;if(month>11){month=0;year++}drawCalendar()});
  drawCalendar();
  document.addEventListener('click',e=>{if(!e.target.closest('.search-area'))closePopups()});
