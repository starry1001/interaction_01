<script>
  const player = document.getElementById('player');
  let playing = false;
  let seconds = 11;
  let timer = null;

  function openPlayer() {
    player.classList.add('open');
    if (!playing) togglePlayback();
  }
  function closePlayer() {
    player.classList.remove('open');
    stopTimer();
  }
  function togglePlayback() {
    playing = !playing;
    document.getElementById('miniPlay').textContent = playing ? 'Ⅱ' : '▶';
    document.getElementById('playerToggle').textContent = playing ? 'Ⅱ' : '▶';
    document.getElementById('playerToggle').setAttribute('aria-label', playing ? 'Pause' : 'Play');
    if (playing) startTimer(); else stopTimer();
  }
  function startTimer() {
    stopTimer();
    timer = setInterval(() => {
      seconds = (seconds + 1) % 90;
      const percent = Math.min(seconds / 90 * 100, 100);
      document.getElementById('progressFill').style.width = percent + '%';
      document.getElementById('progressKnob').style.left = percent + '%';
      const m = Math.floor(seconds / 60), s = seconds % 60;
      document.getElementById('elapsed').textContent = m + ':' + String(s).padStart(2, '0');
    }, 1000);
  }
  function stopTimer() {
    if (timer) { clearInterval(timer); timer = null; }
  }
  function seekTrack(event) {
    const rect = event.currentTarget.getBoundingClientRect();
    const percent = Math.max(0, Math.min(1, (event.clientX - rect.left) / rect.width));
    seconds = Math.round(percent * 90);
    document.getElementById('progressFill').style.width = (percent * 100) + '%';
    document.getElementById('progressKnob').style.left = (percent * 100) + '%';
    const m = Math.floor(seconds / 60), s = seconds % 60;
    document.getElementById('elapsed').textContent = m + ':' + String(s).padStart(2, '0');
  }
  document.addEventListener('keydown', event => {
    if (event.key === 'Escape') closePlayer();
    if (event.code === 'Space' && player.classList.contains('open')) {
      event.preventDefault(); togglePlayback();
    }
  });
</script>
