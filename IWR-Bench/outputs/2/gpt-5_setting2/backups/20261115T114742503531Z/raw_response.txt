<script>
  // Mock "catalog"
  const tracks = [
    { title:"Sacrifice", artist:"Mariah the Scientist", album:"HEARTS SOLD SEPARATELY", length: 210, cover:"__PLACEHOLDER_ASSETS_BASE_DIR__/017.jpg" }
  ];
  let current = 0;
  let isPlaying = false;
  let position = 11; // seconds into the song (to resemble screenshot)
  let timer = null;

  const player = document.getElementById('player');
  const playPause = document.getElementById('playPause');
  const playPauseIcon = document.getElementById('playPauseIcon');
  const elapsedEl = document.getElementById('elapsed');
  const remainingEl = document.getElementById('remaining');
  const barFill = document.getElementById('barFill');

  function fmt(sec){
    sec = Math.max(0, Math.floor(sec));
    return Math.floor(sec/60)+':'+String(sec%60).padStart(2,'0');
  }
  function updateUI(){
    const t = tracks[current];
    const total = t.length;
    const pct = Math.min(100, position/total*100);
    barFill.style.width = pct + '%';
    elapsedEl.textContent = fmt(position);
    remainingEl.textContent = '-'+fmt(total-position);
    playPauseIcon.src = isPlaying ? "__PLACEHOLDER_ASSETS_BASE_DIR__/008.svg" : "__PLACEHOLDER_ASSETS_BASE_DIR__/011.svg";
    // header play icon sync
    const hdrPlay = document.getElementById('hdr-play').querySelector('img');
    hdrPlay.src = isPlaying ? "__PLACEHOLDER_ASSETS_BASE_DIR__/008.svg" : "__PLACEHOLDER_ASSETS_BASE_DIR__/011.svg";
  }
  function startTimer(){
    if(timer) return;
    timer = setInterval(()=>{
      position += 0.5;
      const total = tracks[current].length;
      if(position >= total){ position = total; pause(); }
      updateUI();
    },500);
  }
  function pause(){ isPlaying=false; clearInterval(timer); timer=null; updateUI(); }
  function play(){ isPlaying=true; startTimer(); updateUI(); }

  // Buttons and interactions
  document.getElementById('playPause').addEventListener('click', ()=>{ isPlaying ? pause() : play(); });
  document.getElementById('prev').addEventListener('click', ()=>{ position = Math.max(0, position-10); updateUI(); });
  document.getElementById('next').addEventListener('click', ()=>{ position = Math.min(tracks[current].length, position+10); updateUI(); });

  // Open/close player
  document.getElementById('getStarted').addEventListener('click', ()=>{ player.classList.add('show'); });
  document.getElementById('studentPlan').addEventListener('click', ()=>{ player.classList.add('show'); });
  document.getElementById('openPlayerFromMini').addEventListener('click', ()=>{ player.classList.add('show'); });
  document.getElementById('closePlayer').addEventListener('click', ()=>{ player.classList.remove('show'); });

  // Header play toggles
  document.getElementById('hdr-play').addEventListener('click', ()=>{ isPlaying ? pause() : play(); });
  document.getElementById('hdr-prev').addEventListener('click', ()=>{ position = Math.max(0, position-10); updateUI(); });
  document.getElementById('hdr-next').addEventListener('click', ()=>{ position = Math.min(tracks[current].length, position+10); updateUI(); });

  // Initial UI state to resemble screenshot
  updateUI();
</script>
